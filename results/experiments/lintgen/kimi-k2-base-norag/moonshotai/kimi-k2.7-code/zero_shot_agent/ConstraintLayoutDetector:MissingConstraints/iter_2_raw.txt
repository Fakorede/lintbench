public class ConstraintLayoutDetector extends ResourceXmlDetector {
    private static final String CONSTRAINT_LAYOUT_ANDROIDX =
            "androidx.constraintlayout.widget.ConstraintLayout";
    private static final String CONSTRAINT_LAYOUT_SUPPORT =
            "android.support.constraint.ConstraintLayout";
    private static final String AUTO_URI = "http://schemas.android.com/apk/res-auto";

    private static final String[] HORIZONTAL_CONSTRAINTS = new String[]{
            "layout_constraintStart_toStartOf",
            "layout_constraintStart_toEndOf",
            "layout_constraintLeft_toLeftOf",
            "layout_constraintLeft_toRightOf",
            "layout_constraintEnd_toStartOf",
            "layout_constraintEnd_toEndOf",
            "layout_constraintRight_toLeftOf",
            "layout_constraintRight_toRightOf"
    };
    private static final String[] VERTICAL_CONSTRAINTS = new String[]{
            "layout_constraintTop_toTopOf",
            "layout_constraintTop_toBottomOf",
            "layout_constraintBottom_toTopOf",
            "layout_constraintBottom_toBottomOf",
            "layout_constraintBaseline_toBaselineOf"
    };

    @Override
    public Collection<String> getApplicableElements() {
        return Arrays.asList(CONSTRAINT_LAYOUT_ANDROIDX, CONSTRAINT_LAYOUT_SUPPORT);
    }

    @Override
    public void visitElement(@NonNull XmlContext context, @NonNull Element element) {
        NodeList children = element.getChildNodes();
        for (int i = 0, n = children.getLength(); i < n; i++) {
            Node child = children.item(i);
            if (child.getNodeType() != Node.ELEMENT_NODE) continue;
            Element childElement = (Element) child;
            String tag = childElement.getTagName();
            if (tag.endsWith("Guideline") || tag.endsWith("Barrier")) continue;

            boolean hasHorizontal = hasConstraint(childElement, HORIZONTAL_CONSTRAINTS);
            boolean hasVertical = hasConstraint(childElement, VERTICAL_CONSTRAINTS);
            String width = childElement.getAttributeNS(ANDROID_URI, "layout_width");
            if (width.equals("match_parent") || width.equals("fill_parent")) {
                hasHorizontal = true;
            }
            String height = childElement.getAttributeNS(ANDROID_URI, "layout_height");
            if (height.equals("match_parent") || height.equals("fill_parent")) {
                hasVertical = true;
            }

            if (!hasHorizontal) {
                context.report(ISSUE, childElement, context.getLocation(childElement),
                        "This view is not constrained horizontally: add a constraint from the start or end edge");
            }
            if (!hasVertical) {
                context.report(ISSUE, childElement, context.getLocation(childElement),
                        "This view is not constrained vertically: add a constraint from the top or bottom edge");
            }
        }
    }

    private static boolean hasConstraint(Element element, String[] constraints) {
        for (String constraint : constraints) {
            if (element.hasAttributeNS(AUTO_URI, constraint)) {
                return true;
            }
        }
        return false;
    }

    public static final Issue ISSUE = Issue.create(
            "MissingConstraints",
            "Missing constraints in ConstraintLayout",
            "...",
            Category.CORRECTNESS,
