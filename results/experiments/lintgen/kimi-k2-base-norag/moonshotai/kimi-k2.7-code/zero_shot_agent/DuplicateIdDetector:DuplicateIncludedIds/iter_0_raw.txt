    @Override
    public void visitElement(@NonNull XmlContext context, @NonNull Element element) {
        if (context.getResourceFolderType() != ResourceFolderType.LAYOUT) {
            return;
        }
        FileInfo fileInfo = getFileInfo(context);
        String tag = element.getTagName();
        if (INCLUDE.equals(tag)) {
            String layout = element.getAttribute("layout");
            if (layout != null && layout.startsWith(LAYOUT_PREFIX)) {
                String included = layout.substring(LAYOUT_PREFIX.length());
                String override = stripId(element.getAttributeNS(ANDROID_URI, "id"));
                fileInfo.includes.add(new IncludeInfo(element, included, override,
                        context.getLocation(element)));
            }
            return;
        }
        String id = stripId(element.getAttributeNS(ANDROID_URI, "id"));
        if (id != null && !id.isEmpty()) {
            fileInfo.ids.add(new IdOccurrence(id, context.getLocation(element)));
            if (element == context.getDocument().getDocumentElement()) {
                fileInfo.rootIds.add(id);
            }
        }
    }

    private FileInfo getFileInfo(XmlContext context) {
        String path = context.getFile().getPath();
        FileInfo info = mFileInfos.get(path);
        if (info == null) {
            File file = context.getFile();
            String name = file.getName();
            int dot = name.lastIndexOf('.');
            if (dot > 0) {
                name = name.substring(0, dot);
            }
            info = new FileInfo(file, name);
            mFileInfos.put(path, info);
            List<FileInfo> list = mLayoutInfos.get(name);
            if (list == null) {
                list = new ArrayList<>();
                mLayoutInfos.put(name, list);
            }
            list.add(info);
        }
        return info;
    }

    @Nullable
    private static String stripId(String id) {
        if (id == null || id.isEmpty()) {
            return null;
        }
        return Lint.stripIdPrefix(id);
    }

    @Override
    public void afterCheckProject(@NonNull Context context) {
        for (FileInfo fileInfo : mFileInfos.values()) {
            checkFile(context, fileInfo);
        }
    }

    private void checkFile(Context context, FileInfo fileInfo) {
        Set<String> seen = new HashSet<>();
        for (IdOccurrence occ : fileInfo.ids) {
            seen.add(occ.id);
        }
        for (IncludeInfo include : fileInfo.includes) {
            LayoutResult result = computeLayoutResult(include.layoutName, new HashSet<String>());
            Set<String> incIds = new HashSet<>(result.allIds);
            if (include.overrideId != null) {
                incIds.removeAll(result.rootIds);
                incIds.add(include.overrideId);
            }
            boolean duplicate = result.hasDuplicates;
            List<String> duplicateIds = new ArrayList<>();
            for (String id : incIds) {
                if (!seen.add(id)) {
                    duplicate = true;
                    duplicateIds.add(id);
                }
            }
            if (duplicate) {
                String message = buildMessage(include.layoutName, duplicateIds);
                context.report(ISSUE, include.location, message);
            }
        }
    }

    private String buildMessage(String layoutName, List<String> duplicateIds) {
        StringBuilder sb = new StringBuilder();
        sb.append("The included layout \"@layout/").append(layoutName).append("\"");
        if (!duplicateIds.isEmpty()) {
            sb.append(" contains id");
            if (duplicateIds.size() > 1) {
                sb.append("s");
            }
            sb.append(" (");
            for (int i = 0; i < duplicateIds.size(); i++) {
                if (i > 0) sb.append(", ");
                sb.append(duplicateIds.get(i));
            }
            sb.append(") that conflict");
            if (duplicateIds.size() == 1) {
                sb.append("s");
            }
            sb.append(" with ids already used in this layout");
        } else {
            sb.append(" has duplicate ids");
        }
        return sb.toString();
    }

    @NonNull
    private LayoutResult computeLayoutResult(@NonNull String layoutName, @NonNull Set<String> visiting) {
        LayoutResult cached = mResults.get(layoutName);
        if (cached != null) {
            return cached;
        }
        if (!visiting.add(layoutName)) {
            return new LayoutResult(Collections.<String>emptySet(), Collections.<String>emptySet(), false);
        }
        List<FileInfo> files = mLayoutInfos.get(layoutName);
        if (files == null) {
            visiting.remove(layoutName);
            return new LayoutResult(Collections.<String>emptySet(), Collections.<String>emptySet(), false);
        }
        Set<String> allIds = new HashSet<>();
        Set<String> rootIds = new HashSet<>();
        boolean hasDuplicates = false;
        for (FileInfo file : files) {
            Set<String> fileIds = new HashSet<>();
            // own ids
            for (IdOccurrence occ : file.ids) {
                if (!fileIds.add(occ.id)) {
                    hasDuplicates = true;
                }
                allIds.add(occ.id);
            }
            rootIds.addAll(file.rootIds);
            // includes
            for (IncludeInfo include : file.includes) {
                LayoutResult child = computeLayoutResult(include.layoutName, visiting);
                Set<String> childIds = new HashSet<>(child.allIds);
                if (include.overrideId != null) {
                    childIds.removeAll(child.rootIds);
                    childIds.add(include.overrideId);
                }
                for (String id : childIds) {
                    if (!fileIds.add(id)) {
                        hasDuplicates = true;
                    }
                    allIds.add(id);
                }
            }
        }
        visiting.remove(layoutName);
        LayoutResult result = new LayoutResult(allIds, rootIds, hasDuplicates);
        mResults.put(layoutName, result);
        return result;
    }
}

private LayoutResult computeLayoutResult(String layoutName, Set<String> visiting) {
    LayoutResult cached = mResults.get(layoutName);
    if (cached != null) return cached;
    if (!visiting.add(layoutName)) return new LayoutResult(emptySet, emptySet, false);
    List<FileInfo> files = mLayoutInfos.get(layoutName);
    if (files == null) {
        visiting.remove(layoutName);
        return new LayoutResult(emptySet, emptySet, false);
    }
    Set<String> allIds = new HashSet<>();
    Set<String> rootIds = new HashSet<>();
    boolean hasDuplicates = false;
    for (FileInfo file : files) {
        Set<String> seenInFile = new HashSet<>();
        // own ids
        for (IdOccurrence occ : file.ids) {
            if (!seenInFile.add(occ.id)) hasDuplicates = true;
            allIds.add(occ.id);
        }
        // effective root ids
        Set<String> fileRootIds = new HashSet<>();
        if (file.rootId != null) fileRootIds.add(file.rootId);
        if (file.rootInclude != null) {
            LayoutResult child = computeLayoutResult(file.rootInclude.layoutName, visiting);
            if (file.rootInclude.overrideId != null) {
                fileRootIds.add(file.rootInclude.overrideId);
            } else {
                fileRootIds.addAll(child.rootIds);
            }
        }
        rootIds.addAll(fileRootIds);
        // includes
        for (IncludeInfo include : file.includes) {
            LayoutResult child = computeLayoutResult(include.layoutName, visiting);
            if (child.hasDuplicates) hasDuplicates = true;
            Set<String> childIds = new HashSet<>(child.allIds);
            if (include.overrideId != null) {
                childIds.removeAll(child.rootIds);
                childIds.add(include.overrideId);
            }
            for (String id : childIds) {
                if (!seenInFile.add(id)) hasDuplicates = true;
                allIds.add(id);
            }
        }
    }
    visiting.remove(layoutName);
    LayoutResult result = new LayoutResult(allIds, rootIds, hasDuplicates);
    mResults.put(layoutName, result);
    return result;
}
