package com.android.tools.lint.checks

import com.android.tools.lint.detector.api.*
import java.util.regex.Pattern

class AppBundleLocaleChangesDetector : Detector(), GradleScanner {

    override fun visitFile(context: Context) {
        val file = context.file
        if (file.name != "build.gradle" && file.name != "build.gradle.kts") {
            return
        }

        val contents = context.getContents() ?: return
        val text = String(contents, Charsets.UTF_8)

        // Only analyze application modules
        if (!text.contains("com.android.application")) {
            return
        }

        // Check if language splits are explicitly disabled
        val languageSplitDisabled = Pattern.compile(
            "enableSplit\\s*=\\s*false",
            Pattern.CASE_INSENSITIVE
        ).matcher(text).find()

        if (!languageSplitDisabled) {
            context.report(
                ISSUE,
                Location.create(file),
                "App Bundle language splits are enabled by default. If your app changes locales " +
                "at runtime, you must disable language splits in the `bundle` block or use the " +
                "Play Core library to download additional locales dynamically."
            )
        }
    }

    companion object {
        @JvmField
        val ISSUE: Issue = Issue.create(
            id = "AppBundleLocaleChanges",
            briefDescription = "App Bundle handling of runtime locale changes",
            explanation = """
                When changing locales at runtime (e.g. to provide an in-app language switcher), \
                the Android App Bundle must be configured to not split by locale or the Play Core \
                library must be used to download additional locales at runtime.

                To disable language splits, add the following to your module's `build.gradle` file:
                ```
                android {
                    bundle {
                        language {
                            enableSplit = false
                        }
                    }
                }
                ```
                Reference: https://developer.android.com/guide/app-bundle/configure-base#handling_language_changes
            """.trimIndent(),
            category = Category.CORRECTNESS,
            priority = 6,
            severity = Severity.WARNING,
            implementation = Implementation(
                AppBundleLocaleChangesDetector::class.java,
                Scope.GRADLE_SCOPE
            )
        )
    }
}
