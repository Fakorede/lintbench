    override fun visitDocument(context: XmlContext, document: Document) {
        val root = document.documentElement ?: return

        val applicationElement = getFirstChildElement(root, "application") ?: return

            val activity: Element,
            val intentFilter: Element,
            val hasAction: Boolean,
            val hasCategory: Boolean

                                            hasAction = true
                                        }
                                    }

                                        if (filterChild.getAttributeNS(ANDROID_URI, ATTR_NAME) == WEARABLE_CONFIGURATION_CATEGORY) {
                                            hasCategory = true
                                        }
                                    }
                                }
                            }
                            activityIntentFilters.add(
                                ActivityIntentFilterInfo(child, intentFilter, hasAction, hasCategory)
                            )
                        }
                    }
                }
            }
        }

        val activitiesWithAction = activityIntentFilters.filter { it.hasAction }
        val hasValidActivity = activitiesWithAction.any { info ->

    private fun getFirstChildElement(parent: Element, tagName: String): Element? {
        val children = parent.childNodes
        for (i in 0 until children.length) {
            val child = children.item(i)
            if (child.nodeType == Node.ELEMENT_NODE && (child as Element).tagName == tagName) {

            }
        }
        return null

            val child = children.item(i)
            if (child.nodeType == Node.ELEMENT_NODE) {

            }
        }
    }

            }
        }
        return 1
    }

                    value == WATCH_FACE_EDITOR_ACTION
                ) {
                    return childElement
                }
            }
        }
        return null
