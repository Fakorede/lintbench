package com.android.tools.lint.checks

import com.android.SdkConstants.ANDROID_URI
import com.android.SdkConstants.ATTR_MIN_SDK_VERSION
import com.android.SdkConstants.ATTR_NAME
import com.android.SdkConstants.ATTR_VALUE
import com.android.SdkConstants.TAG_ACTIVITY
import com.android.SdkConstants.TAG_APPLICATION
import com.android.SdkConstants.TAG_INTENT_FILTER
import com.android.SdkConstants.TAG_SERVICE
import com.android.SdkConstants.TAG_USES_SDK
import com.android.tools.lint.detector.api.Category
import com.android.tools.lint.detector.api.Detector
import com.android.tools.lint.detector.api.Implementation
import com.android.tools.lint.detector.api.Issue
import com.android.tools.lint.detector.api.Scope
import com.android.tools.lint.detector.api.Severity
import com.android.tools.lint.detector.api.XmlContext
import com.android.tools.lint.detector.api.XmlScanner
import org.w3c.dom.Element
import org.w3c.dom.Node

class WearableConfigurationActionDetector : Detector(), XmlScanner {

    companion object {
        private const val WATCH_FACE_EDITOR_ACTION =
            "androidx.wear.watchface.editor.action.WATCH_FACE_EDITOR"
        private const val WEARABLE_CONFIGURATION_CATEGORY =
            "com.google.android.wearable.watchface.category.WEARABLE_CONFIGURATION"
        private const val WEARABLE_CONFIGURATION_ACTION_METADATA =
            "com.google.android.wearable.watchface.wearableConfigurationAction"

        val ISSUE = Issue.create(
            id = "WearableConfigurationAction",
            briefDescription = "Wear configuration action metadata must match an activity",
            explanation = """
                Only when a watch face service defines `wearableConfigurationAction` metadata \
                with the value `WATCH_FACE_EDITOR`, there should be an activity in the same \
                package which has an intent filter for `WATCH_FACE_EDITOR` (with \
                `com.google.android.wearable.watchface.category.WEARABLE_CONFIGURATION` if \
                `minSdkVersion` is less than 30).

                See https://developer.android.com/training/wearables/watch-faces/configuration \
                for more details.
            """,
            category = Category.CORRECTNESS,
            priority = 5,
            severity = Severity.ERROR,
            implementation = Implementation(
                WearableConfigurationActionDetector::class.java,
                Scope.MANIFEST_SCOPE
            )
        ).addMoreInfo("https://developer.android.com/training/wearables/watch-faces/configuration")
    }

    override fun getApplicableElements(): Collection<String> = listOf(TAG_APPLICATION)

    override fun visitElement(context: XmlContext, element: Element) {
        if (element.tagName != TAG_APPLICATION) return

        val minSdkVersion = getMinSdkVersion(element)

        // Collect services with wearableConfigurationAction metadata = WATCH_FACE_EDITOR
        // Store the metadata element for reporting
        data class ServiceInfo(val service: Element, val metaDataElement: Element)
        val servicesWithMetadata = mutableListOf<ServiceInfo>()

        // Collect activities with WATCH_FACE_EDITOR intent filter
        // Track whether they have the required category
        data class ActivityInfo(
            val activity: Element,
            val intentFilter: Element,
            val hasAction: Boolean,
            val hasCategory: Boolean
        )
        val activitiesWithWatchFaceEditorAction = mutableListOf<ActivityInfo>()

        val children = element.childNodes
        for (i in 0 until children.length) {
            val child = children.item(i)
            if (child.nodeType != Node.ELEMENT_NODE) continue
            val childElement = child as Element

            when (childElement.tagName) {
                TAG_SERVICE -> {
                    val metaData = findWatchFaceEditorMetadata(childElement)
                    if (metaData != null) {
                        servicesWithMetadata.add(ServiceInfo(childElement, metaData))
                    }
                }
                TAG_ACTIVITY -> {
                    val intentFilterInfo = findWatchFaceEditorIntentFilter(childElement)
                    if (intentFilterInfo != null) {
                        activitiesWithWatchFaceEditorAction.add(
                            ActivityInfo(
                                childElement,
                                intentFilterInfo.first,
                                intentFilterInfo.second,
                                intentFilterInfo.third
                            )
                        )
                    }
                }
            }
        }

        // Determine if there's a fully valid activity (has action + category if needed)
        val hasValidActivity = activitiesWithWatchFaceEditorAction.any { info ->
            info.hasAction && (minSdkVersion >= 30 || info.hasCategory)
        }

        // If services have metadata but no valid activity
        if (servicesWithMetadata.isNotEmpty() && !hasValidActivity) {
            for (serviceInfo in servicesWithMetadata) {
                // Check if there's an activity with action but missing category
                val activityMissingCategory = activitiesWithWatchFaceEditorAction.firstOrNull {
                    it.hasAction && !it.hasCategory && minSdkVersion < 30
                }

                if (activityMissingCategory != null) {
                    // Report on the intent filter that's missing the category
                    context.report(
                        ISSUE,
                        activityMissingCategory.intentFilter,
                        context.getLocation(activityMissingCategory.intentFilter),
                        "This intent filter must also have a category " +
                            "`$WEARABLE_CONFIGURATION_CATEGORY` since `minSdkVersion` < 30"
                    )
                } else {
                    // No activity at all with the action
                    context.report(
                        ISSUE,
                        serviceInfo.metaDataElement,
                        context.getLocation(serviceInfo.metaDataElement),
                        "Watch face service defines `wearableConfigurationAction` metadata " +
                            "but there is no activity in the package with a matching intent " +
                            "filter for `WATCH_FACE_EDITOR`"
                    )
                }
            }
        }

        // If activities have the intent filter but no service with the metadata
        if (activitiesWithWatchFaceEditorAction.isNotEmpty() && servicesWithMetadata.isEmpty()) {
            for (activityInfo in activitiesWithWatchFaceEditorAction) {
                context.report(
                    ISSUE,
                    activityInfo.intentFilter,
                    context.getLocation(activityInfo.intentFilter),
                    "Activity has intent filter for `WATCH_FACE_EDITOR` but there is no " +
                        "watch face service defining `wearableConfigurationAction` metadata " +
                        "with value `WATCH_FACE_EDITOR`"
                )
            }
        }
    }

    private fun getMinSdkVersion(applicationElement: Element): Int {
        val manifest = applicationElement.parentNode as? Element ?: return 1
        val children = manifest.childNodes
        for (i in 0 until children.length) {
            val child = children.item(i)
            if (child.nodeType == Node.ELEMENT_NODE &&
                (child as Element).tagName == TAG_USES_SDK
            ) {
                val minSdk = child.getAttributeNS(ANDROID_URI, ATTR_MIN_SDK_VERSION)
                if (minSdk.isNotEmpty()) {
                    return minSdk.toIntOrNull() ?: 1
                }
            }
        }
        return 1
    }

    private fun findWatchFaceEditorMetadata(service: Element): Element? {
        val children = service.childNodes
        for (i in 0 until children.length) {
            val child = children.item(i)
            if (child.nodeType != Node.ELEMENT_NODE) continue
            val childElement = child as Element
            if (childElement.tagName == "meta-data") {
                val name = childElement.getAttributeNS(ANDROID_URI, ATTR_NAME)
                val value = childElement.getAttributeNS(ANDROID_URI, ATTR_VALUE)
                if (name == WEARABLE_CONFIGURATION_ACTION_METADATA &&
                    value == WATCH_FACE_EDITOR_ACTION
                ) {
                    return childElement
                }
            }
        }
        return null
    }

    /**
     * Returns Triple(intentFilterElement, hasWatchFaceEditorAction, hasWearableConfigurationCategory)
     * or null if no intent filter with WATCH_FACE_EDITOR action is found.
     */
    private fun findWatchFaceEditorIntentFilter(
        activity: Element
    ): Triple<Element, Boolean, Boolean>? {
        val children = activity.childNodes
        for (i in 0 until children.length) {
            val child = children.item(i)
            if (child.nodeType != Node.ELEMENT_NODE) continue
            val childElement = child as Element
            if (childElement.tagName == TAG_INTENT_FILTER) {
                var hasAction = false
                var hasCategory = false

                val filterChildren = childElement.childNodes
                for (j in 0 until filterChildren.length) {
                    val filterChild = filterChildren.item(j)
                    if (filterChild.nodeType != Node.ELEMENT_NODE) continue
                    val filterChildElement = filterChild as Element

                    when (filterChildElement.tagName) {
                        "action" -> {
                            val name = filterChildElement.getAttributeNS(ANDROID_URI, ATTR_NAME)
                            if (name == WATCH_FACE_EDITOR_ACTION) {
                                hasAction = true
                            }
                        }
                        "category" -> {
                            val name = filterChildElement.getAttributeNS(ANDROID_URI, ATTR_NAME)
                            if (name == WEARABLE_CONFIGURATION_CATEGORY) {
                                hasCategory = true
                            }
                        }
                    }
                }

                if (hasAction) {
                    return Triple(childElement, hasAction, hasCategory)
                }
            }
        }
        return null
    }
}
