package com.android.tools.lint.checks

import com.android.SdkConstants
import com.android.tools.lint.detector.api.Category
import com.android.tools.lint.detector.api.Context
import com.android.tools.lint.detector.api.Detector
import com.android.tools.lint.detector.api.GradleContext
import com.android.tools.lint.detector.api.GradleScanner
import com.android.tools.lint.detector.api.Implementation
import com.android.tools.lint.detector.api.Issue
import com.android.tools.lint.detector.api.JavaContext
import com.android.tools.lint.detector.api.Location
import com.android.tools.lint.detector.api.Scope
import com.android.tools.lint.detector.api.Severity
import com.android.tools.lint.detector.api.SourceCodeScanner
import com.intellij.psi.PsiMethod
import org.jetbrains.uast.UCallExpression
import java.util.EnumSet

class AppBundleLocaleChangesDetector : Detector(), SourceCodeScanner, GradleScanner {

    // Track whether the project uses Play Core SplitInstallManager (locale download)
    private var usesPlayCoreSplitInstall = false

    // Track whether bundle { language { enableSplit = false } } is configured
    private var bundleLanguageSplitDisabled = false

    // Track whether we found a locale change call
    private val localeChangeCalls = mutableListOf<Location>()

    // Track whether we found a configuration update call
    private val configurationUpdateCalls = mutableListOf<Location>()

    override fun getApplicableMethodNames(): List<String> = listOf(
        // Locale setting methods
        "setLocale",
        "setLocales",
        "setDefault",
        // Configuration update
        "updateConfiguration",
        "createConfigurationContext",
        // AppCompatDelegate locale methods
        "setApplicationLocales",
        "setDefaultNightMode",
        // Resources update
        "updateResources"
    )

    override fun visitMethodCall(context: JavaContext, node: UCallExpression, method: PsiMethod) {
        val methodName = method.name
        val containingClass = method.containingClass?.qualifiedName ?: return

        when {
            // java.util.Locale.setDefault(Locale)
            methodName == "setDefault" && containingClass == "java.util.Locale" -> {
                localeChangeCalls.add(context.getLocation(node))
            }

            // Configuration.setLocale / setLocales
            methodName == "setLocale" && containingClass == "android.content.res.Configuration" -> {
                localeChangeCalls.add(context.getLocation(node))
            }

            methodName == "setLocales" && containingClass == "android.content.res.Configuration" -> {
                localeChangeCalls.add(context.getLocation(node))
            }

            // Context.createConfigurationContext
            methodName == "createConfigurationContext" &&
                    (containingClass == "android.content.Context" ||
                            containingClass == "android.content.ContextWrapper") -> {
                configurationUpdateCalls.add(context.getLocation(node))
            }

            // Resources.updateConfiguration (deprecated but still used)
            methodName == "updateConfiguration" && containingClass == "android.content.res.Resources" -> {
                localeChangeCalls.add(context.getLocation(node))
            }

            // AppCompatDelegate.setApplicationLocales (AndroidX per-app language)
            methodName == "setApplicationLocales" &&
                    containingClass == "androidx.appcompat.app.AppCompatDelegate" -> {
                localeChangeCalls.add(context.getLocation(node))
            }

            // Play Core SplitInstallManager — user is handling locales via Play Core
            containingClass == "com.google.android.play.core.splitinstall.SplitInstallManager" ||
                    containingClass == "com.google.android.play.core.splitinstall.SplitInstallManagerFactory" -> {
                usesPlayCoreSplitInstall = true
            }
        }
    }

    // -------------------------------------------------------------------------
    // Gradle DSL scanning
    // -------------------------------------------------------------------------

    override fun checkDslPropertyAssignment(
        context: GradleContext,
        property: String,
        value: String,
        parent: String,
        parentParent: String?,
        propertyCookie: Any,
        valueCookie: Any
    ) {
        // We are looking for:
        //   android {
        //     bundle {
        //       language {
        //         enableSplit = false
        //       }
        //     }
        //   }
        if (property == "enableSplit" &&
            parent == "language" &&
            parentParent == "bundle" &&
            value == "false"
        ) {
            bundleLanguageSplitDisabled = true
        }
    }

    // -------------------------------------------------------------------------
    // Report after all files have been processed
    // -------------------------------------------------------------------------

    override fun afterCheckRootProject(context: Context) {
        // If the project has opted out of language splitting or uses Play Core
        // to download locales, there is no issue.
        if (bundleLanguageSplitDisabled || usesPlayCoreSplitInstall) {
            return
        }

        // Report each locale-change call site
        val allCalls = localeChangeCalls + configurationUpdateCalls
        for (location in allCalls) {
            context.report(
                issue = ISSUE,
                location = location,
                message = MESSAGE
            )
        }
    }

    companion object {

        private const val MESSAGE =
            "Found a locale change at runtime. When using Android App Bundles, " +
                    "the system may not have the required locale resources available unless " +
                    "you either disable language splitting " +
                    "(`bundle { language { enableSplit = false } }` in your Gradle file) " +
                    "or use the Play Core library to request the additional language split at runtime. " +
                    "See https://developer.android.com/guide/app-bundle/configure-base#handling_language_changes"

        @JvmField
        val ISSUE: Issue = Issue.create(
            id = "AppBundleLocaleChanges",
            briefDescription = "App Bundle handling of runtime locale changes",
            explanation = """
                When changing locales at runtime (e.g. to provide an in-app language switcher), \
                the Android App Bundle must be configured to not split by locale or the Play Core \
                library must be used to download additional locales at runtime.

                If your app changes the locale at runtime without one of these mitigations, \
                the required string resources for the new locale may not be present on the device \
                because the App Bundle infrastructure only delivers the resources for the locale \
                that was active at install time.

                To fix this, either:
                1. Disable language splitting in your `build.gradle`:
                   ```
                   android {
                       bundle {
                           language {
                               enableSplit = false
                           }
                       }
                   }
                   ```
                2. Use the Play Core library's `SplitInstallManager` to request the language \
                   split at runtime before switching locales.

                See https://developer.android.com/guide/app-bundle/configure-base#handling_language_changes
            """,
            category = Category.CORRECTNESS,
            priority = 6,
            severity = Severity.WARNING,
            implementation = Implementation(
                AppBundleLocaleChangesDetector::class.java,
                EnumSet.of(Scope.JAVA_FILE, Scope.GRADLE_FILE),
                EnumSet.of(Scope.JAVA_FILE),
                EnumSet.of(Scope.GRADLE_FILE)
            )
        )
    }
}
