    }

    @Override
    public void visitElement(@NonNull XmlContext context, @NonNull Element element) {
        // Skip root element
        if (element.getParentNode() == element.getOwnerDocument()) {
            return;
        }

        // Only process direct children of the root element (top-level resources)
        if (element.getParentNode() == null) {
            return;
        }
        org.w3c.dom.Node parent = element.getParentNode();
        if (parent == null || parent.getNodeType() != org.w3c.dom.Node.ELEMENT_NODE) {
            // parent is the document or something else - skip
            // Actually we want direct children of the root <resources> element
            return;
        }

            // This element is nested more than one level deep - skip
            return;
        }

        // Get the resource name from the "name" attribute
        Attr nameAttr = element.getAttributeNode("name");
        if (nameAttr == null) {
            return;
        }

        String name = nameAttr.getValue();
        if (name == null || name.isEmpty()) {
            return;
        }

        // Normalize the name: replace dots with underscores for comparison
        // (dots and underscores are treated as equivalent in resource names)
        String normalizedName = name.replace('.', '_');

        // Determine the resource type
        String tagName = element.getTagName();
        String resourceType = getResourceType(tagName, element);
        if (resourceType == null) {
            return;
        }

        // Build a unique key for this resource: type + normalized name
        String resourceKey = resourceType + ":" + normalizedName;

        // Get the folder path (parent directory of the file)
        File file = context.file;
        File folder = file.getParentFile();
        if (folder == null) {
            return;
        }
        String folderPath = folder.getPath();

        // Look up or create the map for this folder
        Map<String, Location> resourceMap = mFolderToResourceMap.get(folderPath);
        if (resourceMap == null) {
            resourceMap = new HashMap<>();
            mFolderToResourceMap.put(folderPath, resourceMap);
        }

        Location currentLocation = context.getLocation(nameAttr);

        if (resourceMap.containsKey(resourceKey)) {
            // Duplicate found!
            Location previousLocation = resourceMap.get(resourceKey);

            // Create a secondary location pointing to the previous definition
            Location secondary = Location.create(
                    previousLocation.getFile(),
                    previousLocation.getStart(),
                    previousLocation.getEnd());

                    element,
                    currentLocation,
                    String.format(

            resourceMap.put(resourceKey, currentLocation);
        }
