```
package com.android.tools.lint.checks;

import static com.android.SdkConstants.ANDROID_NS_NAME;
import static com.android.SdkConstants.ANDROID_URI;
import static com.android.SdkConstants.ATTR_DEFAULT_VALUE;
import static com.android.SdkConstants.ATTR_KEY;
import static com.android.SdkConstants.ATTR_TITLE;
import static com.android.SdkConstants.TAG_RESTRICTION;
import static com.android.SdkConstants.TAG_RESTRICTIONS;

import com.android.annotations.NonNull;
import com.android.resources.ResourceFolderType;
import com.android.tools.lint.detector.api.Category;
import com.android.tools.lint.detector.api.Detector;
import com.android.tools.lint.detector.api.Implementation;
import com.android.tools.lint.detector.api.Issue;
import com.android.tools.lint.detector.api.Scope;
import com.android.tools.lint.detector.api.Severity;
import com.android.tools.lint.detector.api.XmlContext;
import com.google.common.collect.Sets;
import java.util.Arrays;
import java.util.Collection;
import java.util.HashSet;
import java.util.Set;
import org.w3c.dom.Element;
import org.w3c.dom.Node;

public class RestrictionsDetector extends Detector implements Detector.XmlScanner {
    public static final Issue ISSUE = Issue.create(
            "ValidRestrictions",
            "Invalid Restrictions Descriptor",
            "Ensures that an application's restrictions XML file is properly formed. ...",
            Category.CORRECTNESS,
            6,
            Severity.ERROR,
            new Implementation(RestrictionsDetector.class, Scope.RESOURCE_FILE_SCOPE));

    private static final String ATTR_RESTRICTION_TYPE = "restrictionType";
    private static final String ATTR_ENTRIES = "entries";
    private static final String ATTR_ENTRY_VALUES = "entryValues";
    private static final String TYPE_BUNDLE = "bundle";
    private static final String TYPE_BUNDLE_ARRAY = "bundle_array";
    private static final String TYPE_HIDDEN = "hidden";
    private static final String TYPE_CHOICE = "choice";
    private static final String TYPE_MULTI_SELECT = "multi_select";
    private static final Set<String> VALID_TYPES = Sets.newHashSet(
            "bool", "string", "integer", TYPE_CHOICE, TYPE_MULTI_SELECT,
            TYPE_HIDDEN, TYPE_BUNDLE, TYPE_BUNDLE_ARRAY);
    private static final int MAX_NESTED_RESTRICTIONS = 2;

    @Override
    public boolean appliesTo(@NonNull ResourceFolderType folderType) {
        return folderType == ResourceFolderType.XML;
    }

    @Override
    public Collection<String> getApplicableElements() {
        return Arrays.asList(TAG_RESTRICTIONS, TAG_RESTRICTION);
    }

    @Override
    public void visitElement(@NonNull XmlContext context, @NonNull Element element) {
        String tag = element.getTagName();
        if (TAG_RESTRICTIONS.equals(tag)) {
            checkRestrictions(context, element);
        } else if (TAG_RESTRICTION.equals(tag)) {
            checkRestriction(context, element);
        }
    }

    private static void checkRestrictions(XmlContext context, Element element) {
        // Ensure root element is restrictions
        if (element != element.getOwnerDocument().getDocumentElement()) {
            return;
        }
        Set<String> keys = new HashSet<>();
        Node child = element.getFirstChild();
        while (child != null) {
            if (child.getNodeType() == Node.ELEMENT_NODE) {
                Element childElement = (Element) child;
                String tag = childElement.getTagName();
                if (!TAG_RESTRICTION.equals(tag)) {
                    context.report(ISSUE, childElement,
                            context.getLocation(childElement),
                            "Unexpected element `<" + tag + ">` under `<restrictions>`; expected `<restriction>`");
                } else {
                    String key = childElement.getAttributeNS(ANDROID_URI, ATTR_KEY);
                    if (key.isEmpty()) {
                        key = childElement.getAttribute(ATTR_KEY);
                    }
                    if (!key.isEmpty() && !keys.add(key)) {
                        context.report(ISSUE, childElement,
                                context.getLocation(childElement),
                                "Duplicate restriction key `" + key + "`");
                    }
                }
            }
            child = child.getNextSibling();
        }
    }

    private static void checkRestriction(XmlContext context, Element element) {
        String type = getAttribute(element, ATTR_RESTRICTION_TYPE);
        if (type.isEmpty()) {
            context.report(ISSUE, element, context.getLocation(element),
                    "Missing `android:restrictionType` attribute");
            return;
        } else if (!VALID_TYPES.contains(type)) {
            context.report(ISSUE, element, context.getLocation(element),
                    "Invalid restriction type `" + type + "`");
        }

        String key = getAttribute(element, ATTR_KEY);
        if (key.isEmpty()) {
            context.report(ISSUE, element, context.getLocation(element),
                    "Missing `android:key` attribute");
        } else if (key.startsWith("@")) {
            context.report(ISSUE, element, context.getLocation(element),
                    "Restriction key must be a non-localized string, but was `" + key + "`");
        }

        if (!TYPE_HIDDEN.equals(type)) {
            String title = getAttribute(element, ATTR_TITLE);
            if (title.isEmpty()) {
                context.report(ISSUE, element, context.getLocation(element),
                        "Missing `android:title` attribute");
            }
        }

        if (TYPE_CHOICE.equals(type) || TYPE_MULTI_SELECT.equals(type)) {
            if (getAttribute(element, ATTR_ENTRIES).isEmpty()) {
                context.report(...);
            }
            if (getAttribute(element, ATTR_ENTRY_VALUES).isEmpty()) {
                context.report(...);
            }
        }

        if (TYPE_BUNDLE.equals(type) || TYPE_BUNDLE_ARRAY.equals(type)) {
            if (!getAttribute(element, ATTR_DEFAULT_VALUE).isEmpty()) {
                context.report(..., "A restriction of type `bundle` or `bundle_array` cannot have a default value");
            }

            int count = 0;
            Node child = element.getFirstChild();
            while (child != null) {
                if (child.getNodeType() == Node.ELEMENT_NODE) {
                    Element childElement = (Element) child;
                    String tag = childElement.getTagName();
                    if (!TAG_RESTRICTION.equals(tag)) {
                        context.report(...);
                    } else {
                        count++;
                        if (count > MAX_NESTED_RESTRICTIONS) {
                            context.report(..., "A bundle restriction can contain at most " + MAX_NESTED_RESTRICTIONS + " restrictions");
                        }
                    }
                }
                child = child.getNextSibling();
            }
            if (count == 0) {
                context.report(..., "A restriction of type `" + type + "` must contain at least one nested `<restriction>` element");
            }
        } else {
            // non-bundle: no children allowed
            Node child = element.getFirstChild();
            while (child != null) {
                if (child.getNodeType() == Node.ELEMENT_NODE) {
                    context.report(...);
                }
                child = child.getNextSibling();
            }
        }
    }

    private static String getAttribute(Element element, String name) {
        String value = element.getAttributeNS(ANDROID_URI, name);
        if (value.isEmpty()) {
            value = element.getAttribute(name);
        }
        return value;
    }
}
```

```
    private static void checkRestrictions(XmlContext context, Element element) {
        // Only check the root <restrictions> element
        if (element.getParentNode() != element.getOwnerDocument()) {
            return;
        }
        Set<String> keys = new HashSet<>();
        Node child = element.getFirstChild();
        while (child != null) {
            if (child.getNodeType() == Node.ELEMENT_NODE) {
                Element childElement = (Element) child;
                String tag = childElement.getTagName();
                if (!TAG_RESTRICTION.equals(tag)) {
                    context.report(ISSUE, childElement, context.getLocation(childElement),
                            "Unexpected element `<" + tag + ">` under `<restrictions>`; expected `<restriction>`");
                } else {
                    String key = getAttribute(childElement, ATTR_KEY);
                    if (!key.isEmpty() && !keys.add(key)) {
                        context.report(ISSUE, childElement, context.getLocation(childElement),
                                "Duplicate restriction key `" + key + "`");
                    }
                }
            }
            child = child.getNextSibling();
        }
    }

    private static void checkRestriction(XmlContext context, Element element) {
        String type = getAttribute(element, ATTR_RESTRICTION_TYPE);
        if (type.isEmpty()) {
            context.report(ISSUE, element, context.getLocation(element),
                    "Missing `android:restrictionType` attribute");
            return;
        }
        if (!VALID_TYPES.contains(type)) {
            context.report(ISSUE, element, context.getLocation(element),
                    "Invalid restriction type `" + type + "`");
        }

        String key = getAttribute(element, ATTR_KEY);
        if (key.isEmpty()) {
            context.report(ISSUE, element, context.getLocation(element),
                    "Missing `android:key` attribute");
        } else if (key.startsWith("@")) {
            context.report(ISSUE, element, context.getLocation(element),
                    "Restriction key must be a non-localized string, but was `" + key + "`");
        }

        if (!TYPE_HIDDEN.equals(type)) {
            String title = getAttribute(element, ATTR_TITLE);
            if (title.isEmpty()) {
                context.report(ISSUE, element, context.getLocation(element),
                        "Missing `android:title` attribute");
            }
        }

        if (TYPE_CHOICE.equals(type) || TYPE_MULTI_SELECT.equals(type)) {
            if (getAttribute(element, ATTR_ENTRIES).isEmpty()) {
                context.report(ISSUE, element, context.getLocation(element),
                        "Missing `android:entries` attribute");
            }
            if (getAttribute(element, ATTR_ENTRY_VALUES).isEmpty()) {
                context.report(ISSUE, element, context.getLocation(element),
                        "Missing `android:entryValues` attribute");
            }
        }

        if (TYPE_BUNDLE.equals(type) || TYPE_BUNDLE_ARRAY.equals(type)) {
            if (!getAttribute(element, ATTR_DEFAULT_VALUE).isEmpty()) {
                context.report(ISSUE, element, context.getLocation(element),
                        "A restriction of type `bundle` or `bundle_array` cannot have a default value");
            }

            int count = 0;
            Node child = element.getFirstChild();
            while (child != null) {
                if (child.getNodeType() == Node.ELEMENT_NODE) {
                    Element childElement = (Element) child;
                    String tag = childElement.getTagName();
                    if (!TAG_RESTRICTION.equals(tag)) {
                        context.report(ISSUE, childElement, context.getLocation(childElement),
                                "Unexpected element `<" + tag + ">` under `<restriction>`; expected `<restriction>`");
                    } else {
                        count++;
                        if (count > MAX_NESTED_RESTRICTIONS) {
                            context.report(ISSUE, childElement, context.getLocation(childElement),
                                    "A bundle restriction can contain at most " + MAX_NESTED_RESTRICTIONS + " nested restrictions");
                        }
                    }
                }
                child = child.getNextSibling();
            }
            if (count == 0) {
                context.report(ISSUE, element, context.getLocation(element),
                        "A restriction of type `" + type + "` must contain at least one nested `<restriction>` element");
            }
        } else {
            Node child = element.getFirstChild();
            while (child != null) {
                if (child.getNodeType() == Node.ELEMENT_NODE) {
                    Element childElement = (Element) child;
                    context.report(ISSUE, childElement, context.getLocation(childElement),
                            "A restriction of type `" + type + "` cannot contain child elements");
                }
                child = child.getNextSibling();
            }
        }
    }

    private static String getAttribute(Element element, String name) {
        String value = element.getAttributeNS(ANDROID_URI, name);
        if (value.isEmpty()) {
            value = element.getAttribute(name);
        }
        return value;
    }
```

```
public class RestrictionsDetector extends Detector implements Detector.XmlScanner {
    public static final Issue ISSUE = Issue.create(
            "ValidRestrictions",
            "Invalid Restrictions Descriptor",
            "Ensures that an application's restrictions XML file is properly formed.",
            Category.CORRECTNESS,
            6,
            Severity.ERROR,
            new Implementation(RestrictionsDetector.class, Scope.RESOURCE_FILE_SCOPE));

    private static final String TAG_RESTRICTIONS = "restrictions";
    private static final String TAG_RESTRICTION = "restriction";
    private static final String ATTR_KEY = "key";
    private static final String ATTR_TITLE = "title";
    private static final String ATTR_RESTRICTION_TYPE = "restrictionType";
    private static final String ATTR_ENTRIES = "entries";
    private static final String ATTR_ENTRY_VALUES = "entryValues";
    private static final String ATTR_DEFAULT_VALUE = "defaultValue";
    private static final String TYPE_BUNDLE = "bundle";
    private static final String TYPE_BUNDLE_ARRAY = "bundle_array";
    private static final String TYPE_HIDDEN = "hidden";
    private static final String TYPE_CHOICE = "choice";
    private static final String TYPE_MULTI_SELECT = "multi_select";
    private static final Set<String> VALID_TYPES = ImmutableSet.of(
            "bool", "string", "integer", TYPE_CHOICE, TYPE_MULTI_SELECT,
            TYPE_HIDDEN, TYPE_BUNDLE, TYPE_BUNDLE_ARRAY);
    private static final int MAX_NESTED_RESTRICTIONS = 2;

    @Override
    public boolean appliesTo(@NonNull ResourceFolderType folderType) {
        return folderType == ResourceFolderType.XML;
    }

    @Override
    public Collection<String> getApplicableElements() {
        return Arrays.asList(TAG_RESTRICTIONS, TAG_RESTRICTION);
    }

    @Override
    public void visitElement(@NonNull XmlContext context, @NonNull Element element) {
        String tag = element.getTagName();
        if (TAG_RESTRICTIONS.equals(tag)) {
            checkRestrictions(context, element);
        } else if (TAG_RESTRICTION.equals(tag)) {
            checkRestriction(context, element);
        }
    }

    private static void checkRestrictions(XmlContext context, Element element) {
        // Only validate the root <restrictions> element
        if (element.getParentNode().getNodeType() != Node.DOCUMENT_NODE) {
            return;
        }

        Set<String> keys = new HashSet<>();
        Node child = element.getFirstChild();
        while (child != null) {
            if (child.getNodeType() == Node.ELEMENT_NODE) {
                Element childElement = (Element) child;
                String tag = childElement.getTagName();
                if (!TAG_RESTRICTION.equals(tag)) {
                    context.report(ISSUE, childElement, context.getLocation(childElement),
                            "Unexpected element `<" + tag + ">` under `<restrictions>`; expected `<restriction>`");
                } else {
                    String key = childElement.getAttributeNS(ANDROID_URI, ATTR_KEY);
                    if (!key.isEmpty() && !keys.add(key)) {
                        context.report(ISSUE, childElement, context.getLocation(childElement),
                                "Duplicate restriction key `" + key + "`");
                    }
                }
            }
            child = child.getNextSibling();
        }
    }

    private static void checkRestriction(XmlContext context, Element element) {
        String type = element.getAttributeNS(ANDROID_URI, ATTR_RESTRICTION_TYPE);
        if (type.isEmpty()) {
            context.report(ISSUE, element, context.getLocation(element),
                    "Missing `android:restrictionType` attribute");
            return;
        }
        if (!VALID_TYPES.contains(type)) {
            context.report(ISSUE, element, context.getLocation(element),
                    "Invalid restriction type `" + type + "`");
        }

        String key = element.getAttributeNS(ANDROID_URI, ATTR_KEY);
        if (key.isEmpty()) {
            context.report(ISSUE, element, context.getLocation(element),
                    "Missing `android:key` attribute");
        } else if (key.startsWith("@")) {
            context.report(ISSUE, element, context.getLocation(element),
                    "Restriction key must be a non-localized string");
        }

        if (!TYPE_HIDDEN.equals(type)) {
            String title = element.getAttributeNS(ANDROID_URI, ATTR_TITLE);
            if (title.isEmpty()) {
                context.report(ISSUE, element, context.getLocation(element),
                        "Missing `android:title` attribute");
            }
        }

        if (TYPE_CHOICE.equals(type) || TYPE_MULTI_SELECT.equals(type)) {
            if (element.getAttributeNS(ANDROID_URI, ATTR_ENTRIES).isEmpty()) {
                context.report(ISSUE, element, context.getLocation(element),
                        "Missing `android:entries` attribute");
            }
            if (element.getAttributeNS(ANDROID_URI, ATTR_ENTRY_VALUES).isEmpty()) {
                context.report(ISSUE, element, context.getLocation(element),
                        "Missing `android:entryValues` attribute");
            }
        }

        if (TYPE_BUNDLE.equals(type) || TYPE_BUNDLE_ARRAY.equals(type)) {
            if (!element.getAttributeNS(ANDROID_URI, ATTR_DEFAULT_VALUE).isEmpty()) {
                context.report(ISSUE, element, context.getLocation(element),
                        "A restriction of type `bundle` or `bundle_array` cannot have a default value");
            }

            int count = 0;
            Node child = element.getFirstChild();
            while (child != null) {
                if (child.getNodeType() == Node.ELEMENT_NODE) {
                    Element childElement = (Element) child;
                    String tag = childElement.getTagName();
                    if (!TAG_RESTRICTION.equals(tag)) {
                        context.report(ISSUE, childElement, context.getLocation(childElement),
                                "Unexpected element `<" + tag + ">` under `<restriction>`; expected `<restriction>`");
                    } else {
                        count++;
                        if (count > MAX_NESTED_RESTRICTIONS) {
                            context.report(ISSUE, childElement, context.getLocation(childElement),
                                    "A bundle restriction can contain at most " + MAX_NESTED_RESTRICTIONS + " restrictions");
                        }
                    }
                }
                child = child.getNextSibling();
            }
            if (count == 0) {
                context.report(ISSUE, element, context.getLocation(element),
                        "A restriction of type `" + type + "` must contain at least one nested `<restriction>` element");
            }
        } else {
            Node child = element.getFirstChild();
            while (child != null) {
                if (child.getNodeType() == Node.ELEMENT_NODE) {
                    Element childElement = (Element) child;
                    context.report(ISSUE, childElement, context.getLocation(childElement),
                            "A restriction of type `" + type + "` cannot contain child elements");
                }
                child = child.getNextSibling();
            }
        }
    }
}
```

```kt
package com.android.tools.lint.checks

import com.android.SdkConstants.ANDROID_URI
import com.android.resources.ResourceFolderType
import com.android.tools.lint.detector.api.Category
import com.android.tools.lint.detector.api.Detector
import com.android.tools.lint.detector.api.Implementation
import com.android.tools.lint.detector.api.Issue
import com.android.tools.lint.detector.api.Scope
import com.android.tools.lint.detector.api.Severity
import com.android.tools.lint.detector.api.XmlContext
import org.w3c.dom.Element
import org.w3c.dom.Node

class RestrictionsDetector : Detector(), Detector.XmlScanner {

    override fun appliesTo(folderType: ResourceFolderType): Boolean =
        folderType == ResourceFolderType.XML

    override fun getApplicableElements(): Collection<String> =
        listOf(TAG_RESTRICTIONS, TAG_RESTRICTION)

    override fun visitElement(context: XmlContext, element: Element) {
        when (element.tagName) {
            TAG_RESTRICTIONS -> checkRestrictions(context, element)
            TAG_RESTRICTION -> checkRestriction(context, element)
        }
    }

    private fun checkRestrictions(context: XmlContext, element: Element) {
        if (element != element.ownerDocument.documentElement) {
            return
        }
        val keys = mutableSetOf<String>()
        for (child in element.childElements) {
            if (child.tagName != TAG_RESTRICTION) {
                context.report(
                    ISSUE,
                    child,
                    context.getLocation(child),
                    "Unexpected element `<${child.tagName}>` under `<restrictions>`; expected `<restriction>`"
                )
                continue
            }
            val key = child.attr(ATTR_KEY)
            if (key.isNotEmpty() && !keys.add(key)) {
                context.report(
                    ISSUE,
                    child,
                    context.getLocation(child),
                    "Duplicate restriction key `$key`"
                )
            }
        }
    }

    private fun checkRestriction(context: XmlContext, element: Element) {
        if (element == element.ownerDocument.documentElement) {
            context.report(
                ISSUE,
                element,
                context.getLocation(element),
                "Restrictions must be defined in a `<restrictions>` root element"
            )
            return
        }

        val type = element.attr(ATTR_RESTRICTION_TYPE)
        if (type.isEmpty()) {
            context.report(
                ISSUE,
                element,
                context.getLocation(element),
                "Missing `android:restrictionType` attribute"
            )
            return
        }
        if (type !in VALID_TYPES) {
            context.report(
                ISSUE,
                element,
                context.getLocation(element),
                "Invalid restriction type `$type`"
            )
        }

        val key = element.attr(ATTR_KEY)
        if (key.isEmpty()) {
            context.report(
                ISSUE,
                element,
                context.getLocation(element),
                "Missing `android:key` attribute"
            )
        } else if (key.startsWith("@")) {
            context.report(
                ISSUE,
                element,
                context.getLocation(element),
                "Restriction key must be a non-localized string, but was `$key`"
            )
        }

        if (type != TYPE_HIDDEN) {
            val title = element.attr(ATTR_TITLE)
            if (title.isEmpty()) {
                context.report(
                    ISSUE,
                    element,
                    context.getLocation(element),
                    "Missing `android:title` attribute"
                )
            }
        }

        if (type == TYPE_CHOICE || type == TYPE_MULTI_SELECT) {
            if (element.attr(ATTR_ENTRIES).isEmpty()) {
                context.report(
                    ISSUE,
                    element,
                    context.getLocation(element),
                    "Missing `android:entries` attribute"
                )
            }
            if (element.attr(ATTR_ENTRY_VALUES).isEmpty()) {
                context.report(
                    ISSUE,
                    element,
                    context.getLocation(element),
                    "Missing `android:entryValues` attribute"
                )
            }
        }

        if (type == TYPE_BUNDLE || type == TYPE_BUNDLE_ARRAY) {
```
