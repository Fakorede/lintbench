public class RegistrationDetector extends Detector
        implements Detector.ClassScanner, Detector.XmlScanner {
    private static final String ACTIVITY = "android/app/Activity";
    private static final String SERVICE = "android/app/Service";
    private static final String PROVIDER = "android/content/ContentProvider";

    public static final Issue ISSUE = Issue.create(
            "Registered",
            "Class is not registered in the manifest",
            "Activities, services and content providers should be registered ...",
            Category.CORRECTNESS,
            6,
            Severity.WARNING,
            new Implementation(
                    RegistrationDetector.class,
                    Scope.CLASS_FILE_SCOPE,
                    Scope.MANIFEST_SCOPE));

    private List<ClassNode> mClasses;
    private Set<String> mManifestNames;
    private String mPackage;

    @Override
    public void beforeCheckProject(@NonNull Context context) {
        mClasses = new ArrayList<>();
        mManifestNames = new HashSet<>();
        mPackage = null;
    }

    @Override
    public void afterCheckProject(@NonNull Context context) {
        for (ClassNode classNode : mClasses) {
            String name = classNode.name.replace('/', '.');
            if (!mManifestNames.contains(name)) {
                // determine type
                String component = getComponentType(classNode); // activity/service/provider
                context.report(ISSUE, getLocation(context, classNode),
                    String.format("The <%s> %s is not registered in the manifest", component, name));
            }
        }
    }

    @Override
    public void checkClass(@NonNull ClassContext context, @NonNull ClassNode classNode) {
        if ((classNode.access & Opcodes.ACC_ABSTRACT) != 0) {
            return;
        }
        if ((classNode.access & Opcodes.ACC_INTERFACE) != 0) {
            return;
        }
        if (classNode.name == null) {
            return;
        }
        if (isComponent(context, classNode)) {
            mClasses.add(classNode);
        }
    }

    private boolean isComponent(ClassContext context, ClassNode classNode) {
        ClassNode cls = classNode;
        while (cls != null) {
            if (ACTIVITY.equals(cls.superName)
                    || SERVICE.equals(cls.superName)
                    || PROVIDER.equals(cls.superName)) {
                return true;
            }
            cls = getParentClass(context, cls);
        }
        return false;
    }

    @Nullable
    private ClassNode getParentClass(ClassContext context, ClassNode classNode) {
        String superName = classNode.superName;
        if (superName == null || superName.equals("java/lang/Object")) {
            return null;
        }
        try {
            ClassReader reader = context.getDriver().getClassReader(superName);
            ClassNode node = new ClassNode();
            reader.accept(node, 0);
            return node;
        } catch (IOException e) {
            return null;
        }
    }

    @Override
    public Collection<String> getApplicableElements() {
        return Arrays.asList(TAG_MANIFEST, TAG_ACTIVITY, TAG_SERVICE, TAG_PROVIDER);
    }

    @Override
    public void visitElement(@NonNull XmlContext context, @NonNull Element element) {
        String tag = element.getTagName();
        if (TAG_MANIFEST.equals(tag)) {
            mPackage = element.getAttribute(ATTR_PACKAGE);
            return;
        }
        String name = element.getAttributeNS(ANDROID_URI, ATTR_NAME);
        if (name == null || name.isEmpty()) {
            return;
        }
        if (mPackage == null) {
            mPackage = context.getProject().getPackage();
        }
        mManifestNames.add(resolveClassName(name, mPackage));
    }

    private static String resolveClassName(String name, String packageName) {
        if (name.startsWith(".")) {
            return packageName + name;
        }
        if (name.indexOf('.') == -1) {
            return packageName + '.' + name;
        }
        return name;
    }
}

        final String name;
        final Location location;
        final String type;

        Candidate(String name, Location location, String type) {
            this.name = name;
            this.location = location;
            this.type = type;

    private Set<String> mRegisteredNames;
    private String mManifestPackage;

    public void beforeCheckProject(@NonNull Context context) {
        mCandidates = new ArrayList<>();
        mRegisteredNames = new HashSet<>();
        mManifestPackage = null;
    }

    @Override
    public void afterCheckProject(@NonNull Context context) {
        if (mCandidates == null || mRegisteredNames == null) {
            return;
        }
        for (Candidate candidate : mCandidates) {
            if (!mRegisteredNames.contains(candidate.name)) {
                context.report(
                        ISSUE,
                        candidate.location,
                        "The <" + candidate.type + "> " + candidate.name
                                + " is not registered in the manifest");
            }
        }
    }

    @Override
    public void checkClass(@NonNull ClassContext context, @NonNull ClassNode classNode) {
        if ((classNode.access & Opcodes.ACC_ABSTRACT) != 0) {
            return;
        }
        if ((classNode.access & Opcodes.ACC_INTERFACE) != 0) {
            return;
        }
        if (classNode.name == null) {
            return;
        }

        String type = getComponentType(context, classNode);
        if (type == null) {
            return;
        }

        String name = classNode.name.replace('/', '.');
        Location location = context.getLocation(classNode);
        mCandidates.add(new Candidate(name, location, type));
    }

    @Nullable
    private static String getComponentType(@NonNull ClassContext context, @NonNull ClassNode classNode) {
        ClassNode current = classNode;
        while (current != null && current.superName != null) {
            String superName = current.superName;
            if (superName.equals(ACTIVITY)) {
                return "activity";
            }
            if (superName.equals(SERVICE)) {
                return "service";
            }
            if (superName.equals(PROVIDER)) {
                return "provider";
            }
            current = getSuperClass(context, current);
        }
        return null;
    }

    @Nullable
    private static ClassNode getSuperClass(
            @NonNull ClassContext context, @NonNull ClassNode classNode) {
        String superName = classNode.superName;
        if (superName == null || superName.equals("java/lang/Object")) {
            return null;
        }
        try {
            ClassReader reader = context.getDriver().getClassReader(superName);
            ClassNode node = new ClassNode();
            reader.accept(node, 0);
            return node;
        } catch (IOException e) {
            return null;
        }
    }

    @Override
    public Collection<String> getApplicableElements() {
        return Arrays.asList(TAG_MANIFEST, TAG_ACTIVITY, TAG_SERVICE, TAG_PROVIDER);
    }

    @Override
    public void visitElement(@NonNull XmlContext context, @NonNull Element element) {
        String tag = element.getTagName();
        if (TAG_MANIFEST.equals(tag)) {
            String pkg = element.getAttribute(ATTR_PACKAGE);
            if (pkg != null && !pkg.isEmpty()) {
                mManifestPackage = pkg;
            }
            return;
        }

        String className = element.getAttributeNS(ANDROID_URI, ATTR_NAME);
        if (className == null || className.isEmpty()) {
            return;
        }

        if (mManifestPackage == null || mManifestPackage.isEmpty()) {
            mManifestPackage = context.getProject().getPackage();
        }

        mRegisteredNames.add(resolveClassName(className, mManifestPackage));
    }

    @NonNull
    private static String resolveClassName(@NonNull String name, @Nullable String packageName) {
        if (name.startsWith(".")) {
            if (packageName == null || packageName.isEmpty()) {
                return name;
            }
            return packageName + name;
        }

        if (name.indexOf('.') == -1) {
            if (packageName == null || packageName.isEmpty()) {
                return name;
            }
            return packageName + '.' + name;
        }

        return name;
    }
}

                Project project = context.getProject();
                mCandidates.computeIfAbsent(project, k -> new ArrayList<>()).add(new Candidate(name, location, type));
            }
        };
