@Override
public void checkResourceFolder(@NonNull ResourceContext context) {
    ResourceFolderType folderType = context.getResourceFolderType();
    if (folderType != ResourceFolderType.DRAWABLE) {
        return;
    }

    File folder = context.getFolder();
    String folderName = folder.getName();
    int index = folderName.indexOf('-');
    String density;
    if (index != -1) {
        density = folderName.substring(index + 1);
    } else {
        density = "";
    }

    Density densityEnum = Density.getById(density);
    if (densityEnum == null) {
        return;
    }

    for (File file : context.getFiles()) {
        String name = file.getName();
        if (!isImageFile(name)) {
            continue;
        }
        String base = getBaseName(name);
        Map<Density, File> map = mNames.get(base);
        if (map == null) {
            map = new EnumMap<Density, File>(Density.class);
            mNames.put(base, map);
        }
        map.put(densityEnum, file);
    }
}

package com.android.tools.lint.checks;

import com.android.annotations.NonNull;
import com.android.annotations.Nullable;
import com.android.resources.ResourceFolderType;
import com.android.tools.lint.detector.api.Category;
import com.android.tools.lint.detector.api.Context;
import com.android.tools.lint.detector.api.Detector;
import com.android.tools.lint.detector.api.Implementation;
import com.android.tools.lint.detector.api.Issue;
import com.android.tools.lint.detector.api.Location;
import com.android.tools.lint.detector.api.ResourceContext;
import com.android.tools.lint.detector.api.Scope;
import com.android.tools.lint.detector.api.Severity;

import java.io.File;
import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Locale;
import java.util.Map;
import java.util.Set;

public class IconDetector extends Detector implements Detector.ResourceFolderScanner {

    private static final String INCLUDE_LDPI = "ANDROID_LINT_INCLUDE_LDPI";

    private static final List<String> DENSITY_ORDER = Arrays.asList(
            "ldpi", "mdpi", "hdpi", "xhdpi", "xxhdpi");

    private static final Set<String> KNOWN_DENSITIES = new HashSet<>(DENSITY_ORDER);
    static {
        KNOWN_DENSITIES.add("xxxhdpi");
        KNOWN_DENSITIES.add("nodpi");
        KNOWN_DENSITIES.add("tvdpi");
        KNOWN_DENSITIES.add("anydpi");
    }

    public static final Issue ICON_DENSITIES = Issue.create(
            "IconDensities",
            "Icon densities validation",
            "Icons will look best if a custom version is provided for each of the major screen density classes (low, medium, high, extra high).  This lint check identifies icons which do not have complete coverage  across the densities.\n\n Low density is not really used much anymore, so this check ignores  the ldpi density. To force lint to include it, set the environment variable `ANDROID_LINT_INCLUDE_LDPI=true`. For more information on current density usage, see  https://developer.android.com/about/dashboards",
            Category.ICONS,
            4,
            Severity.WARNING,
            new Implementation(IconDetector.class, Scope.RESOURCE_FOLDER_SCOPE));

    private Map<String, Map<String, File>> mNames;

    @Override
    public void beforeCheckProject(@NonNull Context context) {
        mNames = new HashMap<>();
    }

    @Override
    public void checkResourceFolder(@NonNull ResourceContext context) {
        ResourceFolderType folderType = context.getResourceFolderType();
        if (folderType != ResourceFolderType.DRAWABLE) {
            return;
        }

        String density = getDensity(context.getFolder().getName());
        if (density == null) {
            return;
        }

        File[] files = context.getFiles();
        if (files == null) {
            return;
        }

        for (File file : files) {
            String name = file.getName();
            if (!isImageFile(name)) {
                continue;
            }
            String baseName = getBaseName(name);
            Map<String, File> densityFiles = mNames.get(baseName);
            if (densityFiles == null) {
                densityFiles = new HashMap<>();
                mNames.put(baseName, densityFiles);
            }
            densityFiles.put(density, file);
        }
    }

    @Override
    public void afterCheckProject(@NonNull Context context) {
        if (mNames == null) {
            return;
        }

        boolean includeLdpi = Boolean.parseBoolean(System.getenv(INCLUDE_LDPI));
        Set<String> fullSet = new HashSet<>();
        if (includeLdpi) {
            fullSet.add("ldpi");
        }
        fullSet.add("mdpi");
        fullSet.add("hdpi");
        fullSet.add("xhdpi");
        fullSet.add("xxhdpi");

        for (Map.Entry<String, Map<String, File>> entry : mNames.entrySet()) {
            Set<String> missing = new HashSet<>(fullSet);
            missing.removeAll(entry.getValue().keySet());
            if (!missing.isEmpty()) {
                File file = entry.getValue().values().iterator().next();
                context.report(ICON_DENSITIES, Location.create(file),
                        String.format("The image %1$s only appears in the following density folders: %2$s; it should appear in all of: %3$s",
                                entry.getKey(),
                                formatDensities(entry.getValue().keySet()),
                                formatDensities(fullSet)));
            }
        }

        mNames = null;
    }

    @Nullable
    private static String getDensity(String folderName) {
        int index = folderName.indexOf('-');
        if (index == -1) {
            return "mdpi";
        }
        String[] parts = folderName.substring(index + 1).split("-");
        for (String part : parts) {
            if (KNOWN_DENSITIES.contains(part)) {
                return part;
            }
        }
        return null;
    }

    private static boolean isImageFile(String name) {
        String lower = name.toLowerCase(Locale.US);
        return lower.endsWith(".png") || lower.endsWith(".jpg") || lower.endsWith(".jpeg")
                || lower.endsWith(".gif") || lower.endsWith(".webp");
    }

    @NonNull
    private static String getBaseName(String fileName) {
        int dot = fileName.lastIndexOf('.');
        return dot == -1 ? fileName : fileName.substring(0, dot);
    }

    @NonNull
    private static String formatDensities(Set<String> densities) {
        StringBuilder sb = new StringBuilder();
        for (String density : DENSITY_ORDER) {
            if (densities.contains(density)) {
                if (sb.length() > 0) {
                    sb.append(", ");
                }
                sb.append("drawable-").append(density);
            }
        }
        // Also include any unknown densities in the set? For completeness, but not needed.
        for (String density : densities) {
            if (!DENSITY_ORDER.contains(density)) {
                if (sb.length() > 0) {
                    sb.append(", ");
                }
                sb.append("drawable-").append(density);
            }
        }
        return sb.toString();
    }
}
