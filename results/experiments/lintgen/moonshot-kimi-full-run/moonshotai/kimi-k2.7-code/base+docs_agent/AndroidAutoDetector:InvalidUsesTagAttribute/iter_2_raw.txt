private void checkUsesTag(XmlContext context, Element element) {
    Element parent = (Element) element.getParentNode();
    if (parent == null || !TAG_AUTOMOTIVE_APP.equals(parent.getTagName())) {
        return;
    }
    Attr attr = element.getAttributeNode(ATTR_NAME);
    if (attr == null) {
        context.report(INVALID_USES_TAG_ISSUE, context.getLocation(element),
                "The `<uses>` element must specify a `name` attribute");
        return;
    }
    String value = attr.getValue();
    if (value == null || value.isEmpty() || !VALID_USES_NAMES.contains(value)) {
        context.report(INVALID_USES_TAG_ISSUE, context.getValueLocation(attr),
                "Invalid `name` attribute value `" + value
                        + "`. Valid values are `media`, `notification`, or `sms`.");
    } else if (VALUE_MEDIA.equals(value)) {
        mIsMediaApp = true;
    }
}

private void checkAction(XmlContext context, Element element) {
    Element parent = (Element) element.getParentNode();
    if (parent == null || !TAG_INTENT_FILTER.equals(parent.getTagName())) {
        return;
    }
    Element grandparent = (Element) parent.getParentNode();
    if (grandparent == null) {
        return;
    }
    Attr attr = element.getAttributeNodeNS(ANDROID_URI, ATTR_NAME);
    if (attr == null) {
        attr = element.getAttributeNode(ATTR_NAME);
    }
    if (attr == null) {
        return;
    }
    String value = attr.getValue();
    if (TAG_SERVICE.equals(grandparent.getTagName())
            && ACTION_MEDIA_BROWSER_SERVICE.equals(value)) {
        mHasMediaBrowserServiceAction = true;
    }
    if (ACTION_MEDIA_PLAY_FROM_SEARCH.equals(value)) {
        mHasMediaSearchIntentFilter = true;
    }
}
