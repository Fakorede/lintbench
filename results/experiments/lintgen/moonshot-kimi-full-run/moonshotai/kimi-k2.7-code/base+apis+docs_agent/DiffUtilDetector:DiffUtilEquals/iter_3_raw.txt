    override fun visitClass(context: JavaContext, declaration: UClass) {
        for (method in declaration.methods) {
            if (method.name != "areContentsTheSame") continue
            val body = method.uastBody ?: continue
            val isJava = context.psiFile is PsiJavaFile
            val resultExpressions = collectResultExpressions(body)
            for (expr in resultExpressions) {
                checkExpression(context, expr, isJava)
            }
        }
    }

    private fun collectResultExpressions(body: UExpression): List<UExpression> {
        val returns = mutableListOf<UExpression>()
        body.accept(object : AbstractUastVisitor() {
            override fun visitReturnExpression(node: UReturnExpression): Boolean {
                node.returnExpression?.let { returns.add(it) }
                return true
            }
        })
        if (returns.isEmpty()) {
            returns.add(body)
        }
        return returns
    }

    private fun checkExpression(context: JavaContext, expr: UExpression, isJava: Boolean) {
        var e = expr
        while (e is UParenthesizedExpression) {
            e = e.expression
        }
        if (e is UUnaryExpression && e.operator == UastPrefixOperator.LOGICAL_NOT) {
            e = e.operand
            while (e is UParenthesizedExpression) {
                e = e.expression
            }
        }
        when (e) {
            is UBinaryExpression -> checkBinary(context, e, isJava)
            is UCallExpression -> checkEqualsCall(context, e)
        }
    }

    private fun checkBinary(context: JavaContext, node: UBinaryExpression, isJava: Boolean) {
        when (node.operator) {
            UastBinaryOperator.IDENTITY_EQUALS,
            UastBinaryOperator.IDENTITY_NOT_EQUALS -> {
                report(context, node, "Using identity equality (`===` or `!==`) in `areContentsTheSame`; consider using structural equality")
            }
            UastBinaryOperator.EQUALS,
            UastBinaryOperator.NOT_EQUALS -> {
                if (isJava && isObjectComparison(node)) {
                    report(context, node, "Using identity equality (`==` or `!=`) in `areContentsTheSame`; consider using `equals()`")
                }
            }
            else -> {}
        }
    }

    private fun isObjectComparison(node: UBinaryExpression): Boolean {
        if (hasNullLiteral(node.leftOperand) || hasNullLiteral(node.rightOperand)) {
            return false
        }
        val leftType = node.leftOperand.getExpressionType()
        val rightType = node.rightOperand.getExpressionType()

        if (leftType != null && (isPrimitive(leftType) || isEnum(leftType))) return false
        if (rightType != null && (isPrimitive(rightType) || isEnum(rightType))) return false
        return true
    }

    private fun hasNullLiteral(expr: UExpression): Boolean {
        return expr is ULiteralExpression && expr.isNull

    private fun isPrimitive(type: PsiType?): Boolean = type is PsiPrimitiveType
