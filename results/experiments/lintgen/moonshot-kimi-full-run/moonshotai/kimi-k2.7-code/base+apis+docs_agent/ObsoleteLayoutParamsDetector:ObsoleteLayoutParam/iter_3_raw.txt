```
@Override
public boolean appliesTo(ResourceFolderType folderType) {
    return folderType == ResourceFolderType.LAYOUT;
}

@Override
public Collection<String> getApplicableElements() {
    return XmlScannerConstants.ALL;
}

@Override
public void visitElement(XmlContext context, Element element) {
    Node parentNode = element.getParentNode();
    if (parentNode == null || parentNode.getNodeType() != Node.ELEMENT_NODE) {
        return;
    }
    Element parent = (Element) parentNode;
    String parentTag = parent.getTagName();
    if (parentTag == null) return;

    NamedNodeMap attributes = element.getAttributes();
    for (int i = 0, n = attributes.getLength(); i < n; i++) {
        Attr attr = (Attr) attributes.item(i);
        if (!ANDROID_URI.equals(attr.getNamespaceURI())) continue;
        String name = attr.getLocalName();
        if (name == null) name = attr.getName();
        if (name == null || !name.startsWith(PREFIX_LAYOUT_PARAM)) continue;
        if (name.equals(ATTR_LAYOUT_WIDTH) || name.equals(ATTR_LAYOUT_HEIGHT)) continue; // common
        Set<String> validParents = LAYOUT_PARAMS.get(name);
        if (validParents == null) continue; // unknown param, don't report
        if (!validParents.contains(parentTag)) {
            context.report(ISSUE, attr, context.getValueLocation(attr), 
                String.format("The attribute `%1$s` is only defined for the following layouts: %2$s, but it is used in a `%3$s` layout", name, join(validParents), parentTag));
        }
    }
}
```

```
        Node parentNode = element.getParentNode();
        if (parentNode == null || parentNode.getNodeType() != Node.ELEMENT_NODE) {
            return;

            if (first) first = false;
```
