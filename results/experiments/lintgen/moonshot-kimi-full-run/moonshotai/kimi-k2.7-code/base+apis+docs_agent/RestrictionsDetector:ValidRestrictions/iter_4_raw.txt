public class RestrictionsDetector extends Detector implements XmlScanner {
    private static final String ANDROID_URI = "http://schemas.android.com/apk/res/android";
    private static final String TAG_RESTRICTIONS = "restrictions";
    private static final String TAG_RESTRICTION = "restriction";
    private static final String ATTR_KEY = "key";
    private static final String ATTR_RESTRICTION_TYPE = "restrictionType";
    private static final String ATTR_ENTRIES = "entries";
    private static final String ATTR_ENTRY_VALUES = "entryValues";
    private static final String ATTR_DEFAULT_VALUE = "defaultValue";
    private static final String TYPE_BUNDLE = "bundle";
    private static final String TYPE_BUNDLE_ARRAY = "bundle_array";
    private static final Set<String> VALID_TYPES = Sets.newHashSet(
        "bool", "string", "integer", "choice", "hidden", TYPE_BUNDLE, TYPE_BUNDLE_ARRAY);
    private static final int MAX_NUMBER_OF_NESTED_RESTRICTIONS = 100;
    private static final int MAX_NESTING_DEPTH = 2;

    private Map<Document, Set<String>> mIds;

    @Override
    public boolean appliesTo(ResourceFolderType folderType) {
        return folderType == ResourceFolderType.XML;
    }

    @Override
    public Collection<String> getApplicableElements() {
        return Arrays.asList(TAG_RESTRICTIONS, TAG_RESTRICTION);
    }

    @Override
    public void visitDocument(XmlContext context, Document document) {
        Element root = document.getDocumentElement();
        if (root == null) return;
        if (!TAG_RESTRICTIONS.equals(root.getTagName())) {
            context.report(ISSUE, root, context.getLocation(root),
                "Restrictions XML files must have a root <restrictions> tag");
        } else {
            mIds = new HashMap<>();
            mIds.put(document, new HashSet<String>());
        }
    }

    @Override
    public void visitElement(XmlContext context, Element element) {
        if (TAG_RESTRICTIONS.equals(element.getTagName())) {
            if (element.getParentNode().getNodeType() != Node.DOCUMENT_NODE) {
                context.report(...);
            }
            NodeList children = element.getChildNodes();
            for (int i = 0; i < children.getLength(); i++) {
                Node child = children.item(i);
                if (child.getNodeType() == Node.ELEMENT_NODE && !TAG_RESTRICTION.equals(child.getNodeName())) {
                    context.report(...);
                }
            }
        } else if (TAG_RESTRICTION.equals(element.getTagName())) {
            String key = element.getAttributeNS(ANDROID_URI, ATTR_KEY);
            if (key.isEmpty()) {
                context.report(...);
            } else if (key.startsWith("@")) {
                context.report(...);
            } else {
                Set<String> ids = mIds.get(context.document);
                if (ids != null) {
                    if (ids.contains(key)) {
                        context.report(...);
                    } else {
                        ids.add(key);
                    }
                }
            }

            String type = element.getAttributeNS(ANDROID_URI, ATTR_RESTRICTION_TYPE);
            if (type.isEmpty()) {
                context.report(...);
            } else if (!VALID_TYPES.contains(type)) {
                context.report(...);
