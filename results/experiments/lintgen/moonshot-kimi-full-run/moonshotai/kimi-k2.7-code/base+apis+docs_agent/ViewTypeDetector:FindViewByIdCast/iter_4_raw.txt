    var parent = node.uastParent ?: return null
    while (parent is UParenthesizedExpression) {
        parent = parent.uastParent ?: return null
    }
    return when (parent) {

            if (parent.operator == UastBinaryOperator.ASSIGN && parent.leftOperand != node) {
                parent.leftOperand.getExpressionType()
            } else null
        }
        is UReturnExpression -> {

            method?.returnType
        }

            val index = args.indexOf(node)
            if (index >= 0) {

                if (resolved != null) {

                    if (index < params.size) params[index].type
                    else if (params.isNotEmpty() && resolved.isVarArgs) params.last().type

                } else null
            } else null
        }
        is UBinaryExpressionWithType -> {

                parent.type
            } else null
        }
        else -> null

class ViewTypeDetector : Detector(), SourceCodeScanner {

    override fun getApplicableMethodNames(): List<String> = listOf("findViewById", "requireViewById")

    override fun visitMethodCall(context: JavaContext, node: UCallExpression, method: PsiMethod) {
        if (!isFindViewByIdMethod(context, method)) return
        if (method.typeParameters.isEmpty()) return
        if (node.typeArguments.isNotEmpty()) return
        if (isAlreadyCast(node)) return

        if (getTargetType(node) == null) {
            context.report(
                ISSUE,
                node,
                context.getLocation(node),
                "Add explicit cast to View"
            )
        }
    }

    private fun isFindViewByIdMethod(context: JavaContext, method: PsiMethod): Boolean {
        val evaluator = context.evaluator
        return evaluator.isMemberInSubClassOf(method, "android.app.Activity", false) ||
                evaluator.isMemberInSubClassOf(method, "android.view.View", false) ||
                evaluator.isMemberInSubClassOf(method, "android.app.Fragment", false) ||
                evaluator.isMemberInSubClassOf(method, "android.support.v4.app.Fragment", false) ||
                evaluator.isMemberInSubClassOf(method, "androidx.fragment.app.Fragment", false)
    }

    private fun isAlreadyCast(node: UCallExpression): Boolean {
        var parent = node.uastParent
        while (parent is UParenthesizedExpression) {
            parent = parent.uastParent
        }
        return parent is UBinaryExpressionWithType &&
                parent.operationKind == UastBinaryExpressionWithTypeKind.CAST
    }

    private fun getTargetType(node: UCallExpression): PsiType? {
        var current: UElement = node
        var parent = node.uastParent ?: return null

        while (true) {
            while (parent is UParenthesizedExpression) {
                current = parent
                parent = parent.uastParent ?: return null
            }

            when (parent) {
                is UQualifiedReferenceExpression -> {
                    if (parent.selector == current) {
                        current = parent
                        parent = parent.uastParent ?: return null
                        continue
                    }
                    return null
                }
                is UIfExpression -> {
                    if (current == parent.thenExpression || current == parent.elseExpression) {
                        current = parent
                        parent = parent.uastParent ?: return null
                        continue
                    }
                    return null
                }
                is UVariable -> return parent.type
                is UBinaryExpression -> {
                    return if (parent.operator == UastBinaryOperator.ASSIGN && parent.leftOperand != current) {
                        parent.leftOperand.getExpressionType()
                    } else null
                }
                is UReturnExpression -> {
                    return parent.getParentOfType(UMethod::class.java)?.returnType
                }
                is UCallExpression -> {
                    val args = parent.valueArguments
                    val index = args.indexOf(current)
                    if (index >= 0) {
                        val resolved = parent.resolve()
                        if (resolved != null) {
                            val params = resolved.parameterList.parameters
                            return when {
                                index < params.size -> params[index].type
                                resolved.isVarArgs && params.isNotEmpty() -> params.last().type
                                else -> null
                            }
                        }
                    }
                    return null
                }
                is UBinaryExpressionWithType -> {
                    return if (parent.operationKind == UastBinaryExpressionWithTypeKind.CAST) parent.type else null
                }
                else -> return null
            }
        }
    }

    companion object {
        @JvmField
        val ISSUE = Issue.create(
            id = "FindViewByIdCast",
            briefDescription = "Add Explicit Cast",
            explanation = """
                In Android O, the `findViewById` signature switched to using generics, which means that most of the time you can leave out explicit casts and just assign the result of the `findViewById` call to variables of specific view classes.

                However, due to language changes between Java 7 and 8, this change may cause code to not compile without explicit casts. This lint check looks for these scenarios and suggests casts to be added now such that the code will continue to compile if the language level is updated to 1.8.
            """.trimIndent(),
            category = Category.CORRECTNESS,
            priority = 4,
            severity = Severity.WARNING,
            implementation = Implementation(
                ViewTypeDetector::class.java,
                Scope.JAVA_FILE_SCOPE
            )
        )
    }
}
