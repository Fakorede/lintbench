public class ManifestResourceDetector extends Detector implements XmlScanner {
    private static final Implementation IMPLEMENTATION = new Implementation(
            ManifestResourceDetector.class,
            Scope.MANIFEST_SCOPE);

    public static final Issue ISSUE = Issue.create(
            "ManifestResource",
            "Manifest resources cannot vary across configurations",
            "Resources referenced from the manifest cannot vary across configurations...",
            Category.CORRECTNESS,
            6,
            Severity.WARNING,
            IMPLEMENTATION);

    @Override
    public Collection<String> getApplicableAttributes() { return XmlScannerConstants.ALL; }

    @Override
    public void visitAttribute(XmlContext context, Attr attribute) {
        String value = attribute.getValue();
        if (value == null || value.isEmpty()) return;
        ResourceUrl url = ResourceUrl.parse(value);
        if (url == null || url.isTheme() || url.isFramework()) return;
        if (isExempt(attribute)) return;
        ResourceReference reference = ResourceReference.create(url);
        List<File> files = reference.getFiles(context.getClient(), context.getProject());
        if (files == null || files.size() <= 1) return;
        if (isVersionOnly(files)) return;
        context.report(ISSUE, attribute, context.getValueLocation(attribute), ...);
    }
}

public class ManifestResourceDetector extends Detector implements XmlScanner {
    private static final Implementation IMPLEMENTATION = new Implementation(...);
    public static final Issue ISSUE = Issue.create(...);

    @Nullable private Map<String, List<String>> mResourceQualifiers;

    @Override @Nullable public Collection<String> getApplicableAttributes() { return XmlScannerConstants.ALL; }

    @Override public void visitAttribute(@NotNull XmlContext context, @NotNull Attr attribute) {
        String value = attribute.getValue();
        if (value == null || value.isEmpty()) return;
        ResourceUrl url = ResourceUrl.parse(value);
        if (url == null || url.isTheme() || url.isFramework()) return;
        ResourceType type = url.type;
        String name = url.name;
        if (type == null || name == null) return;
        if (isExempt(attribute.getOwnerElement(), attribute)) return;
        List<String> qualifiers = getQualifiers(context, type, name);
        if (qualifiers == null || qualifiers.size() <= 1) return;
        if (isVersionOnly(qualifiers)) return;
        context.report(ISSUE, attribute, context.getValueLocation(attribute), "...");
    }

    private boolean isExempt(Element element, Attr attribute) { ... }

    private List<String> getQualifiers(XmlContext context, ResourceType type, String name) {
        if (mResourceQualifiers == null) mResourceQualifiers = new HashMap<>();
        String key = type.getName() + "/" + name;
        List<String> cached = mResourceQualifiers.get(key);
        if (cached != null) return cached;
        List<String> qualifiers = computeQualifiers(context, type, name);
        mResourceQualifiers.put(key, qualifiers);
        return qualifiers;
    }

    private List<String> computeQualifiers(XmlContext context, ResourceType type, String name) {
        List<String> qualifiers = new ArrayList<>();
        Project project = context.getProject();
        List<File> resourceFolders = project.getResourceFolders();
        if (resourceFolders == null) return qualifiers;
        String typeName = type.getName();
        boolean isValueType = isValueType(typeName);
        for (File resDir : resourceFolders) {
            File[] dirs = resDir.listFiles();
            if (dirs == null) continue;
            for (File dir : dirs) {
                if (!dir.isDirectory()) continue;
                String dirName = dir.getName();
                if (matchesResourceDir(dirName, typeName, isValueType)) {
                    String qualifiersStr = extractQualifiers(dirName, typeName, isValueType);
                    if (containsResource(context, dir, typeName, name)) {
                        qualifiers.add(qualifiersStr);
                    }
                }
            }
        }
        return qualifiers;
    }
