        String tag = element.getTagName();

        if (SdkConstants.TAG_ITEM.equals(tag)) {
            String expectedType = element.getAttributeNS(null, SdkConstants.ATTR_TYPE);
            if (expectedType == null || expectedType.isEmpty()) {

            return null;
