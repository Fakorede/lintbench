private fun detectCycles(context: Context) {
    val visited = mutableSetOf<NodeKey>()
    val stack = mutableSetOf<NodeKey>()
    val path = mutableListOf<NodeKey>()
    val pathEdges = mutableListOf<Edge>() // edges taken

    fun dfs(node: NodeKey): Boolean {
        if (node in stack) {
            // cycle found
            val cycleStart = path.indexOf(node)
            val cycleNodes = path.subList(cycleStart, path.size)
            val cycleEdges = pathEdges.subList(cycleStart, pathEdges.size)
            reportCycle(context, cycleNodes, cycleEdges)
            return true
        }
        if (node in visited) return false
        visited += node
        stack += node
        path += node
        val edges = graph[node] ?: emptyList()
        for (edge in edges) {
            pathEdges += edge
            dfs(edge.target)
            pathEdges.removeAt(pathEdges.size - 1)
        }
        path.removeAt(path.size - 1)
        stack -= node
        return false
    }

    for (node in graph.keys) {
        dfs(node)
    }
}

    lowlink[v] = currentIndex
    currentIndex++
    stack.add(v)
    onStack += v
    for (edge in graph[v] ?: emptyList()) {
        val w = edge.target

            strongconnect(w)
            lowlink[v] = minOf(lowlink[v]!!, lowlink[w]!!)

            lowlink[v] = minOf(lowlink[v]!!, index[w]!!)

fun findCycle(scc: Set<NodeKey>): Pair<List<NodeKey>, List<Edge>>? {
    val visited = mutableSetOf<NodeKey>()
    val stack = mutableListOf<NodeKey>()
    val edgeStack = mutableListOf<Edge>()
    val sccSet = scc.toSet()
    fun dfs(v: NodeKey): Pair<List<NodeKey>, List<Edge>>? {
        if (v in stack) {
            val i = stack.indexOf(v)
            return stack.subList(i, stack.size) to edgeStack.subList(i, edgeStack.size)
        }
        if (v in visited) return null
        visited += v
        stack += v
        for (edge in graph[v] ?: emptyList()) {
            if (edge.target in sccSet) {
                edgeStack += edge
                val res = dfs(edge.target)
                if (res != null) return res
                edgeStack.removeLast()
            }
        }
        stack.removeLast()
        return null
    }
    for (start in scc) {
        val res = dfs(start)
        if (res != null) return res
    }
    return null
}
