```
private fun collectReferences(context: XmlContext, element: Element, from: Resource, refs: MutableList<Reference>, parseText: Boolean) {
    // Style parent
    if (element.tagName == SdkConstants.TAG_STYLE) {
        val name = element.getAttribute(SdkConstants.ATTR_NAME)
        val explicitParent = element.getAttribute(SdkConstants.ATTR_PARENT)
        if (explicitParent.isBlank()) {
            // implied parent from name prefix
            val dot = name.lastIndexOf('.')
            if (dot > 0) {
                val parentName = name.substring(0, dot)
                refs.add(Reference(from, Resource(ResourceType.STYLE, parentName), context.getLocation(element)))
            }
        } else if (!explicitParent.startsWith("?") && !isFrameworkReference(explicitParent)) {
            addReference(context, from, explicitParent, context.getLocation(element), refs, defaultType = ResourceType.STYLE)
        }
    }

    // Attributes
    for (i in 0 until element.attributes.length) {
        val attr = element.attributes.item(i) as Attr
        if (attr.prefix == "tools" || attr.prefix == "xmlns") continue
        val value = attr.value ?: continue
        if (value.startsWith("?")) continue
        if (!value.startsWith("@")) continue
        if (isFrameworkReference(value)) continue
        addReference(context, from, value, context.getLocation(attr), refs)
    }

    // Text content
    if (parseText && element.childNodes.length == 1 && element.firstChild?.nodeType == Node.TEXT_NODE) {
        val text = element.textContent?.trim().orEmpty()
        if (text.startsWith("@") && !text.startsWith("@?") && !isFrameworkReference(text)) {
            addReference(context, from, text, context.getLocation(element), refs)
        }
    }

    // Children
    for (i in 0 until element.childNodes.length) {
        val child = element.childNodes.item(i)
        if (child is Element) {
            collectReferences(context, child, from, refs, parseText)
        }
    }
}
```

```
private fun detectCycles(context: Context) {
    if (graph.isEmpty()) return

    val index = mutableMapOf<Resource, Int>()
    val lowlink = mutableMapOf<Resource, Int>()
    val stack = ArrayDeque<Resource>()
    val onStack = mutableSetOf<Resource>()
    var nextIndex = 0
    val sccs = mutableListOf<List<Resource>>()

    fun strongconnect(v: Resource) {
        index[v] = nextIndex
        lowlink[v] = nextIndex
        nextIndex++
        stack.addLast(v)
        onStack.add(v)

        for (ref in graph[v].orEmpty()) {
            val w = ref.to
            if (w !in graph) continue
            if (w !in index) {
                strongconnect(w)
                lowlink[v] = minOf(lowlink.getValue(v), lowlink.getValue(w))
            } else if (w in onStack) {
                lowlink[v] = minOf(lowlink.getValue(v), index.getValue(w))
            }
        }

        if (lowlink.getValue(v) == index.getValue(v)) {
            val component = mutableListOf<Resource>()
            while (true) {
                val w = stack.removeLast()
                onStack.remove(w)
                component.add(w)
                if (w == v) break
            }
            if (component.size > 1 || graph[component.single()]?.any { it.to == component.single() } == true) {
                sccs.add(component)
            }
        }
    }

    for (v in graph.keys) {
        if (v !in index) strongconnect(v)
    }

    for (scc in sccs) {
        reportScc(context, scc)
    }
}
```

```
private fun reportScc(context: Context, scc: List<Resource>) {
    val sccSet = scc.toSet()
    val start = scc.minByOrNull { "${it.type.getName()}/${it.name}" } ?: return
    val path = mutableListOf<Resource>()
    val pathSet = mutableSetOf<Resource>()

    fun dfs(v: Resource): Boolean {
        path.add(v)
        pathSet.add(v)

        val refs = graph[v].orEmpty().sortedBy { "${it.to.type.getName()}/${it.to.name}" }
        for (ref in refs) {
            val w = ref.to
            if (w !in sccSet) continue

            if (w == start && path.size > 1) {
                val cycle = (path + w).joinToString(" -> ") { "${it.type.getName()}/${it.name}" }
                context.report(ISSUE, ref.location, "Cycle in resource definitions: $cycle")
                return true
            }

            if (w !in pathSet) {
                if (dfs(w)) return true
            }
        }

        path.removeLast()
        pathSet.remove(v)
        return false
    }

    dfs(start)
}
```
