public class DuplicateIdDetector extends Detector implements XmlScanner {
    /** Map from layout name to the set of id's used by the layout */
    private Map<String, Set<String>> mIds;
    /** Map from layout name to the set of layouts it includes */
    private Map<String, Set<String>> mIncludes;
    /** Map from layout name to the file defining it */
    private Map<String, File> mFiles;

    public static final Issue DUPLICATE_INCLUDED_IDS = Issue.create(...);

    @Override
    public boolean appliesTo(ResourceFolderType folderType) {
        return folderType == ResourceFolderType.LAYOUT;
    }

    @Override
    public Collection<String> getApplicableElements() {
        return Arrays.asList(INCLUDE, VIEW_TAG_MERGE);
    }

    @Override
    public void visitElement(XmlContext context, Element element) {
        String layout = getBaseName(context.file.getName());
        mFiles.put(layout, context.file);
        if (element.getTagName().equals(INCLUDE)) {
            Attr attr = element.getAttributeNode(ATTR_LAYOUT);
            if (attr != null) {
                String value = attr.getValue();
                if (value.startsWith(LAYOUT_PREFIX)) {
                    String target = value.substring(LAYOUT_PREFIX.length());
                    Set<String> includes = mIncludes.get(layout);
                    if (includes == null) {
                        includes = new HashSet<>();
                        mIncludes.put(layout, includes);
                    }
                    includes.add(target);
                }
            }
        } else {
            Attr attr = element.getAttributeNode(ATTR_ID);
            if (attr != null) {
                String id = attr.getValue();
                if (id != null && id.startsWith(NEW_ID_PREFIX)) {
                    id = id.substring(id.lastIndexOf('/') + 1);
                    Set<String> ids = mIds.get(layout);
                    if (ids == null) { ids = new HashSet<>(); mIds.put(layout, ids); }
                    ids.add(id);
                }
            }
        }
    }

    @Override
    public void afterCheckEachProject(Context context) {
        if (mIds == null || mIds.isEmpty()) return;
        for (Map.Entry<String, Set<String>> entry : mIncludes.entrySet()) {
            String layout = entry.getKey();
            Set<String> includes = entry.getValue();
            Set<String> checked = new HashSet<>();
            for (String include : includes) {
                checkIncluded(context, layout, include, checked);
            }
        }
    }

    private void checkIncluded(Context context, String layout, String include, Set<String> checked) {
        if (!checked.add(include)) return;
        Set<String> ids = mIds.get(include);
        if (ids != null) {
            Set<String> parentIds = mIds.get(layout);
            if (parentIds != null) {
                for (String id : parentIds) {
                    if (ids.contains(id)) {
                        File file = mFiles.get(layout);
                        context.report(DUPLICATE_INCLUDED_IDS, Location.create(file), String.format("Duplicate id ...", id, include, layout));
                    }
                }
            }
        }
        Set<String> childIncludes = mIncludes.get(include);
        if (childIncludes != null) {
            for (String child : childIncludes) {
                checkIncluded(context, layout, child, checked);
            }
        }
    }
}

public class DuplicateIdDetector extends Detector implements XmlScanner {
    private Map<String, Set<String>> mIds;
    private Map<String, Set<String>> mIncludes;

    @Override
    public boolean appliesTo(@NonNull ResourceFolderType folderType) {
        return folderType == ResourceFolderType.LAYOUT;
    }

    @Override
    @NonNull
    public Collection<String> getApplicableElements() {
        return Arrays.asList(INCLUDE, VIEW_TAG_MERGE);
    }

    @Override
    public void visitElement(@NonNull XmlContext context, @NonNull Element element) {
        String layout = getBaseName(context.file.getName());
        if (element.getTagName().equals(INCLUDE)) {
            Attr attr = element.getAttributeNode(ATTR_LAYOUT);
            if (attr != null) {
                String value = attr.getValue();
                if (value.startsWith(LAYOUT_PREFIX)) {
                    Set<String> includes = mIncludes.get(layout);
                    if (includes == null) {
                        includes = new HashSet<>();
                        mIncludes.put(layout, includes);
                    }
                    includes.add(value.substring(LAYOUT_PREFIX.length()));
                }
            }
        } else {
            Attr attr = element.getAttributeNode(ATTR_ID);
            if (attr != null) {
                String id = attr.getValue();
                if (id != null && id.startsWith(NEW_ID_PREFIX)) {
                    id = id.substring(id.lastIndexOf('/') + 1);
                    Set<String> ids = mIds.get(layout);
                    if (ids == null) { ids = new HashSet<>(); mIds.put(layout, ids); }
                    ids.add(id);
                }
            }
        }
    }

        if (mIds == null) {
