```java
@Override
public void visitAttribute(XmlContext context, Attr attribute) {
    if (!ANDROID_URI.equals(attribute.getNamespaceURI())) return;
    String name = attribute.getLocalName();
    if (name == null || !name.startsWith(PREFIX_LAYOUT_PARAM)) return;
    if (ATTR_LAYOUT_WIDTH.equals(name) || ATTR_LAYOUT_HEIGHT.equals(name)) return;

    Element element = attribute.getOwnerElement();
    if (element == null) return;
    Element parent = getParentElement(element);
    if (parent == null) return;
    String parentTag = parent.getTagName();
    if (parentTag == null) return;
    String simpleName = getSimpleName(parentTag);

    Set<String> valid = getValidParams(simpleName);
    if (valid == null) {
        // unknown layout; maybe custom view suffix
        valid = getValidParamsBySuffix(simpleName);
        if (valid == null) return;
    }

    if (!valid.contains(name)) {
        context.report(ISSUE, attribute, context.getValueLocation(attribute), ...);
    }
}
```

```java
            return;
```

```java
        if (name == null || !name.startsWith(PREFIX_LAYOUT_PARAM)) {
            return;

        if ("layout_width".equals(name) || "layout_height".equals(name)) {
            return;

        if (element == null) {
            return;
        }

        Node parentNode = element.getParentNode();
        if (parentNode == null) {
            return;
        }

        if (parentNode instanceof Element) {
            Element parent = (Element) parentNode;
            String parentTag = parent.getTagName();
            if (parentTag == null) {
                return;
            }
            String simpleName = parentTag.contains(".")
                    ? parentTag.substring(parentTag.lastIndexOf('.') + 1)
                    : parentTag;

                if (isValidConstraintAttribute(name)) {
                    return;
                }
                report(context, attribute, simpleName);
                return;

            String resolved = resolveLayoutName(simpleName);
            if (resolved == null) {
                return;
            }

            Set<String> valid = VALID_PARAMS.get(resolved);
            if (valid == null || valid.contains(name)) {
                return;
            }

            report(context, attribute, resolved);
        } else {
            // Root element: layout params have no effect without a parent layout
            report(context, attribute, null);
        }

    private static boolean isValidConstraintAttribute(@NonNull String name) {
        return name.startsWith("layout_constraint")
                || name.startsWith("layout_goneMargin")
                || name.startsWith("layout_editor_absolute")
                || name.startsWith("layout_margin")
                || "layout_constrainedWidth".equals(name)
                || "layout_constrainedHeight".equals(name)
                || "layout_wrapBehaviorInParent".equals(name);
    }

    @Nullable
    private static String resolveLayoutName(@NonNull String simpleName) {
        if (VALID_PARAMS.containsKey(simpleName)) {
            return simpleName;
        }
        for (String name : LAYOUT_NAMES) {
            if (simpleName.endsWith(name)) {
                return name;
            }
        }
        return null;
    }

    private static void report(@NonNull XmlContext context, @NonNull Attr attribute,
            @Nullable String parentTag) {
        String name = attribute.getLocalName();
        String message;
        if (parentTag != null) {
            message = "Obsolete layout param: `" + name + "` is not defined for `"
                    + parentTag + "`";
        } else {
            message = "Obsolete layout param: `" + name + "` has no effect outside a layout";
        }
        context.report(ISSUE, attribute, context.getValueLocation(attribute), message);
    }
```
