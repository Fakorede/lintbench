        String pkg = context.getMainProject().getPackage();
        if (pkg == null) {
            return name;

        if (name.startsWith(".")) {
            return pkg + name;
        }
        if (name.indexOf('.') == -1) {
            return pkg + "." + name;
        }
        return name;
