public class StateListDetector extends ResourceXmlDetector {
    public static final Issue ISSUE = Issue.create(
        "StateListReachable",
        "Unreachable state in selector",
        "In a `<selector>`, only the last child should omit a state qualifier. ...",
        Category.CORRECTNESS, 5, Severity.WARNING,
        new Implementation(StateListDetector.class, Scope.RESOURCE_FILE_SCOPE));

    private static final String SELECTOR = "selector";
    private static final String ITEM = "item";
    private static final String STATE_PREFIX = "state_";

    @Override public boolean appliesTo(ResourceFolderType folderType) {
        return folderType == ResourceFolderType.DRAWABLE || folderType == ResourceFolderType.COLOR;
    }

    @Override public Collection<String> getApplicableElements() {
        return Collections.singletonList(SELECTOR);
    }

    @Override public void visitElement(@NonNull XmlContext context, @NonNull Element element) {
        List<Element> items = new ArrayList<>();
        Node child = element.getFirstChild();
        while (child != null) {
            if (child.getNodeType() == Node.ELEMENT_NODE && ITEM.equals(child.getNodeName())) {
                items.add((Element) child);
            }
            child = child.getNextSibling();
        }
        if (items.size() < 2) return;
        Element last = items.get(items.size()-1);
        for (Element item : items) {
            if (item == last) continue;
            if (!hasStateQualifier(item)) {
                context.report(ISSUE, item, context.getLocation(item),
                    "This item is not reachable: previous item has no state qualifiers");
            }
        }
    }

    private static boolean hasStateQualifier(Element item) {
        NamedNodeMap attrs = item.getAttributes();
        for (int i=0; i<attrs.getLength(); i++) {
            Attr attr = (Attr) attrs.item(i);
            String name = attr.getLocalName();
            if (name == null) name = attr.getName();
            if (name != null && name.startsWith(STATE_PREFIX)) return true;
        }
        return false;
    }
}
