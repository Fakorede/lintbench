    private static final String JOB_SCHEDULER_CLASS = "android.app.job.JobScheduler";
    private static final String JOB_SERVICE_CLASS = "android.app.job.JobService";
    private static final String BIND_JOB_SERVICE_PERMISSION = "android.permission.BIND_JOB_SERVICE";
    private static final String ANDROID_NS = "http://schemas.android.com/apk/res/android";

        final JavaContext context;
        final UCallExpression call;
        final Location location;
        final String serviceClassName;

        PendingCheck(JavaContext context, UCallExpression call, Location location,

            this.context = context;
            this.call = call;
            this.location = location;
            this.serviceClassName = serviceClassName;
        }
    }

    @Override
    public void afterCheckRootProject(@NonNull Context context) {
        for (PendingCheck pending : mPendingChecks) {
            String serviceClassName = pending.serviceClassName;
            JavaContext javaContext = pending.context;
            JavaEvaluator evaluator = javaContext.getEvaluator();

            // Check 1: Does the class extend JobService?
            PsiClass serviceClass = evaluator.findClass(serviceClassName);
            if (serviceClass != null) {
                boolean extendsJobService = evaluator.extendsClass(
                        serviceClass, JOB_SERVICE_CLASS, false);
                if (!extendsJobService) {
                    javaContext.report(
                            ISSUE,
                            pending.call,
                            pending.location,
                            "Scheduled job class `" + serviceClassName +
                            "` must extend `android.app.job.JobService`"
                    );
                    continue;
                }
            }

            // Check 2: Is the service registered in the manifest?
            String manifestKey = findManifestKey(serviceClassName);
            if (manifestKey == null) {
                javaContext.report(
                        ISSUE,
                        pending.call,
                        pending.location,
                        "Did not find a registered `<service>` with the name `" +
                        serviceClassName + "`"
                );
                continue;
            }

            // Check 3: Does the manifest registration require BIND_JOB_SERVICE?
            String permission = mManifestServices.get(manifestKey);
            if (permission == null || !BIND_JOB_SERVICE_PERMISSION.equals(permission)) {
                javaContext.report(
                        ISSUE,
                        pending.call,
                        pending.location,
                        "The service `" + serviceClassName + "` requires the permission " +
                        "`" + BIND_JOB_SERVICE_PERMISSION + "`, add " +
                        "`android:permission=\"" + BIND_JOB_SERVICE_PERMISSION + "\"` " +
                        "to the `<service>` tag"
                );
            }
        }
    }

    @Nullable
    private String findManifestKey(String serviceClassName) {
        if (mManifestServices.containsKey(serviceClassName)) {
            return serviceClassName;
        }
        // Try matching by simple name suffix
        String simpleName = getSimpleName(serviceClassName);
        for (String key : mManifestServices.keySet()) {
            if (key.endsWith("." + simpleName) || key.equals(simpleName)) {
                return key;
            }
        }
        return null;
    }

    @Nullable

    private static String resolveClassName(String name, String packageName) {
        if (name.startsWith(".")) {
            return packageName + name;
        } else if (!name.contains(".")) {
            if (packageName.isEmpty()) {
                return name;
            }
            return packageName + "." + name;
        }
        return name;
    }

    private static String getSimpleName(String className) {
        int dot = className.lastIndexOf('.');
        if (dot >= 0) {
            return className.substring(dot + 1);
        }
        return className;
    }

    /**
     * UAST visitor that looks for JobInfo.Builder constructor calls to extract the service
     * class name (the second argument to the Builder constructor is a ComponentName, and
     * we look for ComponentName constructor calls within that).
     */
    private static class ServiceClassFinder extends AbstractUastVisitor {
        private final JavaContext mContext;
        private String mServiceClassName;

        ServiceClassFinder(JavaContext context) {
            mContext = context;
        }

        @Nullable
        String getServiceClassName() {
            return mServiceClassName;
        }

        @Override
        public boolean visitCallExpression(@NonNull UCallExpression node) {
            if (mServiceClassName != null) {
                return false;
            }

            PsiMethod resolvedMethod = node.resolve();
            if (resolvedMethod == null) {
                return true;
            }

            PsiClass containingClass = resolvedMethod.getContainingClass();
            if (containingClass == null) {
                return true;
            }

            String qualifiedName = containingClass.getQualifiedName();

            // Look for ComponentName constructor: new ComponentName(context, MyService.class)
            // or new ComponentName(packageName, "com.example.MyService")
            if ("android.content.ComponentName".equals(qualifiedName)) {
                List<UExpression> args = node.getValueArguments();
                if (args.size() >= 2) {
                    UExpression classArg = args.get(1);
                    String className = extractClassName(classArg);
                    if (className != null) {
                        mServiceClassName = className;
                    }
                }
                return false;
            }

            // Also look for JobInfo.Builder constructor which takes (int, ComponentName)
            if ("android.app.job.JobInfo.Builder".equals(qualifiedName) ||
                    "android.app.job.JobInfo$Builder".equals(qualifiedName)) {
                List<UExpression> args = node.getValueArguments();
                if (args.size() >= 2) {
                    UExpression componentNameArg = args.get(1);
                    ServiceClassFinder subFinder = new ServiceClassFinder(mContext);
                    componentNameArg.accept(subFinder);
                    if (subFinder.getServiceClassName() != null) {
                        mServiceClassName = subFinder.getServiceClassName();
                    }
                }
                return false;
            }

            // For build() and other chained calls, continue visiting children
            return false;
        }

        @Nullable
        private String extractClassName(@NonNull UExpression expression) {
            // Handle MyService.class (UClassLiteralExpression)
            if (expression instanceof UClassLiteralExpression) {
                UClassLiteralExpression classLiteral = (UClassLiteralExpression) expression;
                PsiType type = classLiteral.getType();
                if (type instanceof PsiClassType) {
                    PsiClass psiClass = ((PsiClassType) type).resolve();
                    if (psiClass != null) {
                        return psiClass.getQualifiedName();
                    }
                }
            }

            // Handle string literal
            Object value = expression.evaluate();
            if (value instanceof String) {
                return (String) value;
            }

            return null;
        }
    }
