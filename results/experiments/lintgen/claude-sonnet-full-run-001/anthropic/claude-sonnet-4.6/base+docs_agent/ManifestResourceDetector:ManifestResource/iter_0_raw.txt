package com.android.tools.lint.checks;

import static com.android.SdkConstants.ANDROID_MANIFEST_XML;
import static com.android.SdkConstants.ANDROID_URI;
import static com.android.SdkConstants.ATTR_ICON;
import static com.android.SdkConstants.ATTR_LABEL;
import static com.android.SdkConstants.ATTR_NAME;
import static com.android.SdkConstants.ATTR_ROUND_ICON;
import static com.android.SdkConstants.TAG_APPLICATION;
import static com.android.SdkConstants.TAG_USES_SDK;

import com.android.annotations.NonNull;
import com.android.annotations.Nullable;
import com.android.resources.ResourceFolderType;
import com.android.resources.ResourceType;
import com.android.tools.lint.detector.api.Category;
import com.android.tools.lint.detector.api.Context;

import org.w3c.dom.Element;

import java.io.File;
import java.util.ArrayList;

/**
 * Checks for resource references in the manifest that are not allowed to vary
 * by configuration (other than by API version).
 */
public class ManifestResourceDetector extends Detector implements XmlScanner {

    /** The main issue discovered by this detector */
    public static final Issue ISSUE = Issue.create(
            "ManifestResource",
            "Manifest Resource References",
            "Elements in the manifest can reference resources, but those resources cannot " +
            "vary across configurations (except as a special case, by version, and except " +
            "for a few specific package attributes such as the application title and icon).",
            Category.CORRECTNESS,
            5,
            Severity.FATAL,
            new Implementation(
                    ManifestResourceDetector.class,
                    Scope.MANIFEST_AND_RESOURCE_SCOPE,
                    Scope.MANIFEST_SCOPE));

    /**
     * Attributes on the application element that are allowed to vary by configuration
     * (e.g. label, icon, roundIcon, banner, etc.)
     */
    private static final List<String> APPLICATION_ALLOWED_VARYING_ATTRS = Arrays.asList(
            ATTR_LABEL,
            ATTR_ICON,
            ATTR_ROUND_ICON,
            "banner",
            "logo",
            "description"
    );

    /**
     * Map from resource name to the list of resource folder types it appears in.
     * Used to detect configuration-specific resources.
     */
    private final Map<String, List<String>> mResourceFolders = new HashMap<>();

    /**
     * Pending attribute references to check after we've scanned all resource folders.
     */
    private final List<PendingCheck> mPendingChecks = new ArrayList<>();

    /** Whether we're in a manifest-only scope */
    private boolean mManifestOnlyScope = false;

    @Override
    public void beforeCheckRootProject(@NonNull Context context) {
        mResourceFolders.clear();
        mPendingChecks.clear();
    }

    @Override
    public Collection<String> getApplicableElements() {
        return XmlScanner.ALL;
    }

    @Override
    public void visitElement(@NonNull XmlContext context, @NonNull Element element) {
        // Only process manifest files
        if (!context.file.getName().equals(ANDROID_MANIFEST_XML)) {
            return;
        }

        NamedNodeMap attributes = element.getAttributes();
        if (attributes == null) {
            return;
        }

        String tagName = element.getTagName();

        for (int i = 0, n = attributes.getLength(); i < n; i++) {
            Node attribute = attributes.item(i);
            if (!(attribute instanceof Attr)) {
                continue;
            }
            Attr attr = (Attr) attribute;
            String value = attr.getValue();

            // Check if this is a resource reference
            if (value == null || !value.startsWith("@")) {
                continue;
            }

            // Skip @+id/ references
            if (value.startsWith("@+")) {
                continue;
            }

            // Parse the resource reference
            String resourceRef = value.substring(1); // strip leading @
            if (resourceRef.startsWith("android:")) {
                // Framework resources are fine
                continue;
            }

            // Strip optional package prefix (e.g., "com.example:")
            int colonIndex = resourceRef.indexOf(':');
            if (colonIndex != -1) {
                // Check if it's a package-qualified reference (not android:)
                resourceRef = resourceRef.substring(colonIndex + 1);
            }

            // Parse type and name
            int slashIndex = resourceRef.indexOf('/');
            if (slashIndex == -1) {
                continue;
            }

            String resourceTypeName = resourceRef.substring(0, slashIndex);
            String resourceName = resourceRef.substring(slashIndex + 1);

            ResourceType resourceType = ResourceType.fromClassName(resourceTypeName);
            if (resourceType == null) {
                continue;
            }

            // Determine if this attribute is allowed to vary
            boolean allowedToVary = isAllowedToVary(tagName, attr);

            if (!allowedToVary) {
                // We need to check if this resource has configuration-specific variants
                // Store for later checking (after resource folders are scanned)
                mPendingChecks.add(new PendingCheck(
                        context.getLocation(attr),
                        resourceType,
                        resourceName,
                        value,
                        attr.getLocalName()
                ));
            }
        }
    }

    /**
     * Determines if an attribute in the manifest is allowed to reference
     * configuration-varying resources.
     */
    private boolean isAllowedToVary(@NonNull String tagName, @NonNull Attr attr) {
        // The application element's label, icon, roundIcon, etc. are allowed to vary
        if (TAG_APPLICATION.equals(tagName)) {
            String localName = attr.getLocalName();
            if (APPLICATION_ALLOWED_VARYING_ATTRS.contains(localName)) {
                return true;
            }
        }
        return false;
    }

    @Override
    public void afterCheckRootProject(@NonNull Context context) {
        // Now check all pending resource references
        for (PendingCheck check : mPendingChecks) {
            checkResourceVariation(context, check);
        }
    }

    private void checkResourceVariation(@NonNull Context context, @NonNull PendingCheck check) {
        // Look through all resource folders to see if this resource has
        // configuration-specific variants
        List<File> resourceFolders = context.getProject().getResourceFolders();

        boolean hasDefaultDefinition = false;
        boolean hasConfigSpecificDefinition = false;
        String configDescription = null;

        String expectedFolderName = getFolderName(check.resourceType);
        if (expectedFolderName == null) {
            return;
        }

        for (File resDir : resourceFolders) {
            File[] folders = resDir.listFiles();
            if (folders == null) {
                continue;
            }

            for (File folder : folders) {
                String folderName = folder.getName();

                // Check if this folder is of the right type
                int dashIndex = folderName.indexOf('-');
                String baseFolderName = dashIndex == -1 ? folderName : folderName.substring(0, dashIndex);

                if (!baseFolderName.equals(expectedFolderName)) {
                    continue;
                }

                // Check if the resource file exists in this folder
                if (resourceExistsInFolder(folder, check.resourceName, check.resourceType)) {
                    if (dashIndex == -1) {
                        // Default folder (no configuration qualifier)
                        hasDefaultDefinition = true;
                    } else {
                        // Configuration-specific folder
                        String qualifier = folderName.substring(dashIndex + 1);

                        // Version qualifiers (v21, v23, etc.) are allowed
                        if (isVersionQualifierOnly(qualifier)) {
                            // This is fine - version-specific resources are allowed
                            continue;
                        }

                        hasConfigSpecificDefinition = true;
                        configDescription = folderName;
                    }
                }
            }
        }

        if (hasConfigSpecificDefinition) {
            String message = String.format(
                    "Resources referenced from the manifest cannot vary by configuration " +
                    "(except for version qualifiers, e.g. `-v21`). Found resource `%1$s` in " +
                    "configuration-specific folder `%2$s`",
                    check.value,
                    configDescription);

            context.report(ISSUE, check.location, message);
        }
    }

    /**
     * Returns true if the qualifier string represents only a version qualifier (e.g., "v21").
     */
    private static boolean isVersionQualifierOnly(@NonNull String qualifier) {
        // A version qualifier looks like "v21", "v23", etc.
        // It could also be combined like "v21-v23" but we check for simple cases
        // Split by '-' and check each part
        String[] parts = qualifier.split("-");
        for (String part : parts) {
            if (!part.matches("v\\d+")) {
                return false;
            }
        }
        return true;
    }

    /**
     * Returns true if a resource with the given name exists in the given folder.
     */
    private static boolean resourceExistsInFolder(
            @NonNull File folder,
            @NonNull String resourceName,
            @NonNull ResourceType resourceType) {

        File[] files = folder.listFiles();
        if (files == null) {
            return false;
        }

        // For values folders, we'd need to parse XML - for simplicity,
        // check file-based resources (drawables, layouts, etc.)
        ResourceFolderType folderType = ResourceFolderType.getFolderType(folder.getName());
        if (folderType == ResourceFolderType.VALUES) {
            // For values, check if any XML file might contain the resource
            // This is a simplified check - in practice we'd need to parse the XML
            for (File file : files) {
                String fileName = file.getName();
                // Check if the file name (without extension) matches
                int dotIndex = fileName.lastIndexOf('.');
                String baseName = dotIndex == -1 ? fileName : fileName.substring(0, dotIndex);
                if (baseName.equals(resourceName)) {
                    return true;
                }
            }
            // For values, we can't easily check without parsing - return false for now
            return false;
        } else {
            // For file-based resources, check if a file with the resource name exists
            for (File file : files) {
                String fileName = file.getName();
                int dotIndex = fileName.lastIndexOf('.');
                String baseName = dotIndex == -1 ? fileName : fileName.substring(0, dotIndex);
                if (baseName.equals(resourceName)) {
                    return true;
                }
            }
            return false;
        }
    }

    /**
     * Returns the base folder name for a given resource type.
     */
    @Nullable
    private static String getFolderName(@NonNull ResourceType resourceType) {
        ResourceFolderType folderType = ResourceFolderType.getTypeByName(resourceType.getName());
        if (folderType != null) {
            return folderType.getName();
        }
        // Handle special cases
        switch (resourceType) {
            case STRING:
            case BOOL:
            case INTEGER:
            case DIMEN:
            case COLOR:
            case ARRAY:
            case STYLE:
            case ATTR:
            case PLURALS:
                return ResourceFolderType.VALUES.getName();
            default:
                return null;
        }
    }

    /** Holds information about a pending resource reference check */
    private static class PendingCheck {
        final Location location;
        final ResourceType resourceType;
        final String resourceName;
        final String value;
        final String attributeName;

        PendingCheck(
                @NonNull Location location,
                @NonNull ResourceType resourceType,
                @NonNull String resourceName,
                @NonNull String value,
                @NonNull String attributeName) {
            this.location = location;
            this.resourceType = resourceType;
            this.resourceName = resourceName;
            this.value = value;
            this.attributeName = attributeName;
        }
    }
}
