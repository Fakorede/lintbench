        return false

                name.contains("Dialog", ignoreCase = true) ||

                name == "android:Theme.Translucent" ||
                name == "android:Theme.Translucent.NoTitleBar" ||
                name == "android:Theme.Translucent.NoTitleBar.Fullscreen"
    }

    private fun isFixedOrientation(orientation: String): Boolean {
        return FIXED_ORIENTATIONS.contains(orientation)
    }

    private fun stripThemePrefix(theme: String): String {
        return theme
            .removePrefix("@android:style/")
            .removePrefix("@style/")
            .removePrefix("@*android:style/")
            .trim()
    }
