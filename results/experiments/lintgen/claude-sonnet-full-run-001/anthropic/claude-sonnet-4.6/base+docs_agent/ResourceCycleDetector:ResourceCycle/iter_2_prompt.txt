    private fun checkElementForReferences(

        element: Element,
        type: ResourceType,
        resourceName: String

            }

            }
        }

    /**
     * Strip a resource reference prefix like "@style/", "@color/", "@drawable/" etc.
     * Returns null if the text is not a reference to the given type.
     */
    private fun stripResourcePrefix(text: String, type: ResourceType): String? {
        if (text.isBlank()) return null
        if (!text.startsWith("@")) return null
        val withoutAt = text.removePrefix("@").removePrefix("+")
        val typeName = type.getName()
        val prefix = "$typeName/"
        return if (withoutAt.startsWith(prefix)) {
            withoutAt.removePrefix(prefix).trim()
        } else {
            // Also handle namespace-qualified references like "android:color/..."
            val slashIdx = withoutAt.indexOf('/')
            if (slashIdx >= 0) {
                val refType = withoutAt.substring(0, slashIdx).substringAfterLast(':')
                if (refType == typeName) {
                    withoutAt.substring(slashIdx + 1).trim()
                } else null
            } else null
        }
    }

        val locs = locations.getOrPut(type) { mutableMapOf() }
        if (!locs.containsKey(name)) {
            locs[name] = location
        }

    private fun detectCycle(
        context: Context,
        type: ResourceType,
        graph: Map<String, List<String>>,
        node: String,
        visited: MutableSet<String>,
        inStack: MutableSet<String>,
        path: MutableList<String>,
        reported: MutableSet<String>

        inStack.add(node)
        path.add(node)

        val neighbors = graph[node] ?: emptyList()
        for (neighbor in neighbors) {
            if (neighbor !in visited) {
                detectCycle(context, type, graph, neighbor, visited, inStack, path, reported)
            } else if (neighbor in inStack) {
                // Found a cycle — report it
                val cycleKey = buildCycleKey(path, neighbor)
                if (cycleKey !in reported) {
                    reported.add(cycleKey)
                    reportCycle(context, type, path, neighbor)
                }
            }
        }

        path.removeAt(path.size - 1)
        inStack.remove(node)
    }

    private fun buildCycleKey(path: List<String>, cycleStart: String): String {
        val cycleIndex = path.indexOf(cycleStart)
        val cyclePath = if (cycleIndex >= 0) {
            path.subList(cycleIndex, path.size)
        } else {
            path
        }
        return cyclePath.sorted().joinToString(",")
    }

    private fun reportCycle(
        context: Context,
        type: ResourceType,
        path: List<String>,
        cycleStart: String
    ) {
        val cycleIndex = path.indexOf(cycleStart)
        val cyclePath = if (cycleIndex >= 0) {
            path.subList(cycleIndex, path.size) + cycleStart
        } else {
            path + cycleStart
        }

        val typeName = type.getName()
        val cycleDescription = cyclePath.joinToString(" => ") { "@$typeName/$it" }

        val locs = locations[type]
        val location = locs?.get(cycleStart) ?: locs?.values?.firstOrNull()

        val message = "Cycle detected: $cycleDescription"

        if (location != null) {
            context.report(ISSUE, location, message)
        }
    }
