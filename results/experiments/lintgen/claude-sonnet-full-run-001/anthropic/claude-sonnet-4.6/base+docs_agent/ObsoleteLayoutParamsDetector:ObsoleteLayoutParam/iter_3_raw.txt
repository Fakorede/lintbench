package com.android.tools.lint.checks;

import com.android.annotations.NonNull;
import com.android.annotations.Nullable;
import com.android.tools.lint.detector.api.Category;
import com.android.tools.lint.detector.api.Implementation;
import com.android.tools.lint.detector.api.Issue;
import com.android.tools.lint.detector.api.LayoutDetector;
import com.android.tools.lint.detector.api.Scope;
import com.android.tools.lint.detector.api.Severity;
import com.android.tools.lint.detector.api.XmlContext;
import com.android.tools.lint.detector.api.XmlScanner;

import org.w3c.dom.Attr;
import org.w3c.dom.Element;
import org.w3c.dom.NamedNodeMap;
import org.w3c.dom.Node;

import java.util.Arrays;
import java.util.Collection;
import java.util.Collections;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Map;
import java.util.Set;

import static com.android.SdkConstants.ANDROID_URI;

/**
 * Checks for layout params that are not valid for the given parent layout.
 */
public class ObsoleteLayoutParamsDetector extends LayoutDetector implements XmlScanner {

    /** The main issue discovered by this detector */
    public static final Issue ISSUE = Issue.create(
            "ObsoleteLayoutParam",
            "Obsolete layout params",
            "The given layout_param is not defined for the given layout, meaning it has no " +
            "effect. This usually happens when you change the parent layout or move view " +
            "code around without updating the layout params. This will cause useless " +
            "attribute processing at runtime, and is misleading for others reading the " +
            "layout so the parameter should be removed.",
            Category.PERFORMANCE,
            6,
            Severity.WARNING,
            new Implementation(
                    ObsoleteLayoutParamsDetector.class,
                    Scope.RESOURCE_FILE_SCOPE));

    // Common layout params available to all views in any layout
    private static final Set<String> COMMON_PARAMS = new HashSet<>(Arrays.asList(
            "layout_width",
            "layout_height",
            "layout_margin",
            "layout_marginLeft",
            "layout_marginRight",
            "layout_marginTop",
            "layout_marginBottom",
            "layout_marginStart",
            "layout_marginEnd",
            "layout_marginHorizontal",
            "layout_marginVertical"
    ));

    // Params specific to LinearLayout children
    private static final Set<String> LINEAR_LAYOUT_PARAMS = new HashSet<>(Arrays.asList(
            "layout_weight",
            "layout_gravity"
    ));

    // Params specific to RelativeLayout children
    private static final Set<String> RELATIVE_LAYOUT_PARAMS = new HashSet<>(Arrays.asList(
            "layout_above",
            "layout_below",
            "layout_toLeftOf",
            "layout_toRightOf",
            "layout_toStartOf",
            "layout_toEndOf",
            "layout_alignTop",
            "layout_alignBottom",
            "layout_alignLeft",
            "layout_alignRight",
            "layout_alignStart",
            "layout_alignEnd",
            "layout_alignBaseline",
            "layout_alignParentTop",
            "layout_alignParentBottom",
            "layout_alignParentLeft",
            "layout_alignParentRight",
            "layout_alignParentStart",
            "layout_alignParentEnd",
            "layout_centerInParent",
            "layout_centerHorizontal",
            "layout_centerVertical",
            "layout_alignWithParentIfMissing"
    ));

    // Params specific to GridLayout children
    private static final Set<String> GRID_LAYOUT_PARAMS = new HashSet<>(Arrays.asList(
            "layout_row",
            "layout_rowSpan",
            "layout_rowWeight",
            "layout_column",
            "layout_columnSpan",
            "layout_columnWeight",
            "layout_gravity"
    ));

    // Params specific to TableRow children
    private static final Set<String> TABLE_ROW_PARAMS = new HashSet<>(Arrays.asList(
            "layout_column",
            "layout_span",
            "layout_gravity"
    ));

    // Params specific to TableLayout children (rows inside TableLayout)
    private static final Set<String> TABLE_LAYOUT_PARAMS = new HashSet<>(Arrays.asList(
            "layout_gravity"
    ));

    // Params specific to FrameLayout children
    private static final Set<String> FRAME_LAYOUT_PARAMS = new HashSet<>(Arrays.asList(
            "layout_gravity"
    ));

    // Params specific to AbsoluteLayout children
    private static final Set<String> ABSOLUTE_LAYOUT_PARAMS = new HashSet<>(Arrays.asList(
            "layout_x",
            "layout_y"
    ));

    /**
     * Map from layout tag name to the set of layout params that are valid for children
     * of that layout (in addition to the common params).
     */
    private static final Map<String, Set<String>> LAYOUT_TO_PARAMS;

    static {
        LAYOUT_TO_PARAMS = new HashMap<>();
        LAYOUT_TO_PARAMS.put("LinearLayout", LINEAR_LAYOUT_PARAMS);
        LAYOUT_TO_PARAMS.put("RelativeLayout", RELATIVE_LAYOUT_PARAMS);
        LAYOUT_TO_PARAMS.put("GridLayout", GRID_LAYOUT_PARAMS);
        LAYOUT_TO_PARAMS.put("TableRow", TABLE_ROW_PARAMS);
        LAYOUT_TO_PARAMS.put("TableLayout", TABLE_LAYOUT_PARAMS);
        LAYOUT_TO_PARAMS.put("FrameLayout", FRAME_LAYOUT_PARAMS);
        LAYOUT_TO_PARAMS.put("AbsoluteLayout", ABSOLUTE_LAYOUT_PARAMS);
        // RadioGroup extends LinearLayout
        LAYOUT_TO_PARAMS.put("RadioGroup", LINEAR_LAYOUT_PARAMS);
        // ScrollView extends FrameLayout
        LAYOUT_TO_PARAMS.put("ScrollView", FRAME_LAYOUT_PARAMS);
        LAYOUT_TO_PARAMS.put("HorizontalScrollView", FRAME_LAYOUT_PARAMS);
        // ConstraintLayout - no specific layout params beyond common ones
        LAYOUT_TO_PARAMS.put("ConstraintLayout", Collections.<String>emptySet());
        // merge tag
        LAYOUT_TO_PARAMS.put("merge", Collections.<String>emptySet());
    }

    /**
     * All known layout-specific params (union of all layout-specific param sets).
     * These are the params that we check — if a param is in this set but not valid
     * for the current parent, we flag it.
     */
    private static final Set<String> ALL_SPECIFIC_PARAMS;

    static {
        ALL_SPECIFIC_PARAMS = new HashSet<>();
        for (Set<String> params : LAYOUT_TO_PARAMS.values()) {
            ALL_SPECIFIC_PARAMS.addAll(params);
        }
    }

    /**
     * Set of known layout container tags. When a parent is one of these, we know
     * exactly what params are valid. For unknown parents (custom views), we skip.
     */
    private static final Set<String> KNOWN_LAYOUTS = new HashSet<>(LAYOUT_TO_PARAMS.keySet());

    /** Constructs a new {@link ObsoleteLayoutParamsDetector} */
    public ObsoleteLayoutParamsDetector() {
    }

    @Override
    @Nullable
    public Collection<String> getApplicableElements() {
        // Return null to visit all elements
        return ALL;
    }

    @Override
    public void visitElement(@NonNull XmlContext context, @NonNull Element element) {
        // Get the parent element to determine the layout type
        Node parentNode = element.getParentNode();
        if (parentNode == null || parentNode.getNodeType() != Node.ELEMENT_NODE) {
            // Root element or no parent — no layout params to check
            return;
        }

        Element parent = (Element) parentNode;
        String parentTag = parent.getTagName();

        // Get the simple tag name (strip package prefix)
        String simpleParentTag = getSimpleTagName(parentTag);

        // For custom views, we can't know what params they accept, so we only check known layouts
        if (!KNOWN_LAYOUTS.contains(simpleParentTag)) {
            return;
        }

        Set<String> validSpecificParams = LAYOUT_TO_PARAMS.get(simpleParentTag);
        if (validSpecificParams == null) {
            validSpecificParams = Collections.emptySet();
        }

        // Check each attribute of the element
        NamedNodeMap attributes = element.getAttributes();
        for (int i = 0, n = attributes.getLength(); i < n; i++) {
            Attr attr = (Attr) attributes.item(i);
            String name = attr.getLocalName();
            String namespace = attr.getNamespaceURI();

            if (!ANDROID_URI.equals(namespace)) {
                continue;
            }

            if (name == null || !name.startsWith("layout_")) {
                continue;
            }

            // Skip common params — they're always valid
            if (COMMON_PARAMS.contains(name)) {
                continue;
            }

            // Only flag params that are known to be layout-specific
            if (!ALL_SPECIFIC_PARAMS.contains(name)) {
                continue;
            }

            // Check if this param is valid for the current parent layout
            if (!validSpecificParams.contains(name)) {
                context.report(
                        ISSUE,
                        attr,
                        context.getLocation(attr),
                        String.format(
                                "Invalid layout param `%1$s` (not defined by parent layout `%2$s`)",
                                name, simpleParentTag));
            }
        }
    }

    /**
     * Returns the simple tag name (strips package prefix like "android.widget.").
     */
    @NonNull
    private static String getSimpleTagName(@NonNull String tag) {
        int dotIndex = tag.lastIndexOf('.');
        if (dotIndex >= 0) {
            return tag.substring(dotIndex + 1);
        }
        return tag;
    }
}
