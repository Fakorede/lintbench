package com.android.tools.lint.checks

import com.android.tools.lint.detector.api.Category
import com.android.tools.lint.detector.api.Detector
import com.android.tools.lint.detector.api.Implementation
import com.android.tools.lint.detector.api.Issue
import com.android.tools.lint.detector.api.JavaContext
import com.android.tools.lint.detector.api.Scope
import com.android.tools.lint.detector.api.Severity
import com.android.tools.lint.detector.api.SourceCodeScanner
import com.intellij.psi.PsiMethod
import com.intellij.psi.PsiType
import org.jetbrains.uast.UCallExpression
import org.jetbrains.uast.UExpression
import org.jetbrains.uast.ULocalVariable
import org.jetbrains.uast.UReturnExpression
import org.jetbrains.uast.UBinaryExpressionWithType
import org.jetbrains.uast.UastBinaryExpressionWithTypeKind
import org.jetbrains.uast.getParentOfType

class ViewTypeDetector : Detector(), SourceCodeScanner {

    companion object {
        @JvmField
        val ISSUE: Issue = Issue.create(
            id = "FindViewByIdCast",
            briefDescription = "Add Explicit Cast",
            explanation = """
                In Android O, the `findViewById` signature switched to using generics, which \
                means that most of the time you can leave out explicit casts and just assign \
                the result of the `findViewById` call to variables of specific view classes.

                However, due to language changes between Java 7 and 8, this change may cause \
                code to not compile without explicit casts. This lint check looks for these \
                scenarios and suggests casts to be added now such that the code will \
                continue to compile if the language level is updated to 1.8.
                """,
            category = Category.CORRECTNESS,
            priority = 9,
            severity = Severity.WARNING,
            implementation = Implementation(
                ViewTypeDetector::class.java,
                Scope.JAVA_FILE_SCOPE
            )
        )

        private val FIND_VIEW_BY_ID_METHODS = listOf("findViewById", "findViewWithTag", "findViewByPredicate", "findViewsWithText")
        private const val FIND_VIEW_BY_ID = "findViewById"
    }

    override fun getApplicableMethodNames(): List<String> = listOf(FIND_VIEW_BY_ID)

    override fun visitMethodCall(context: JavaContext, node: UCallExpression, method: PsiMethod) {
        // Only care about findViewById calls
        val methodName = node.methodName ?: return
        if (methodName != FIND_VIEW_BY_ID) return

        // Check if this call is already wrapped in an explicit cast
        val castExpression = node.getParentOfType<UBinaryExpressionWithType>(strict = true)
        if (castExpression != null &&
            castExpression.operationKind == UastBinaryExpressionWithTypeKind.TypeCast.INSTANCE) {
            // Already has an explicit cast, no need to warn
            return
        }

        // Check if the result is assigned to a local variable with a specific type
        val localVariable = node.getParentOfType<ULocalVariable>(strict = true)
        if (localVariable != null) {
            val variableType = localVariable.type
            val expressionType = node.getExpressionType()

            if (expressionType != null && variableType != expressionType) {
                // The variable has a more specific type than the return type of findViewById
                // Check if this is a Java file (Kotlin handles this differently)
                val file = context.psiFile ?: return
                val fileName = file.name
                if (fileName.endsWith(".java")) {
                    // Check if the variable type is a subtype of View
                    val variableTypeName = variableType.canonicalText
                    if (variableTypeName != "android.view.View" &&
                        !variableTypeName.startsWith("java.") &&
                        !variableTypeName.startsWith("kotlin.")) {
                        reportIssue(context, node, variableTypeName)
                    }
                }
            }
            return
        }

        // Check if used in a return statement
        val returnExpression = node.getParentOfType<UReturnExpression>(strict = true)
        if (returnExpression != null) {
            val file = context.psiFile ?: return
            val fileName = file.name
            if (fileName.endsWith(".java")) {
                // The return type might be more specific
                val expressionType = node.getExpressionType()
                if (expressionType != null) {
                    val typeName = expressionType.canonicalText
                    if (typeName != "android.view.View") {
                        reportIssue(context, node, typeName)
                    }
                }
            }
        }
    }

    private fun reportIssue(context: JavaContext, node: UCallExpression, targetType: String) {
        val simpleName = targetType.substringAfterLast('.')
        context.report(
            ISSUE,
            node,
            context.getLocation(node),
            "Add explicit cast here; won't compile with Java 8 without it: `($simpleName) ${node.asSourceString()}`"
        )
    }
}
