/**
 * Checks for resources referenced from the manifest that vary by configuration.
 */
public class ManifestResourceDetector extends Detector implements XmlScanner {

    /** The main issue discovered by this detector */
    public static final Issue ISSUE = Issue.create(
            "ManifestResource",
            "Manifest Resource References",
            "Elements in the manifest can reference resources, but those resources cannot " +
            "vary across configurations (except as a special case, by version, and except " +
            "for a few specific package attributes such as the application title and icon).",
            Category.CORRECTNESS,
            8,
            Severity.FATAL,
            new Implementation(
                    ManifestResourceDetector.class,
                    Scope.MANIFEST_SCOPE));

    /**
     * Attributes that are allowed to vary by configuration (locale, density, etc.)
     * for the application element.
     */
    private static final List<String> APPLICATION_ALLOWED_VARYING_ATTRS = Arrays.asList(
            ATTR_LABEL,
            ATTR_ICON,
            ATTR_ROUND_ICON,
            ATTR_LOGO,
            ATTR_DESCRIPTION,
            ATTR_THEME
    );

    /**
     * Attributes that are allowed to vary by configuration for component elements
     * (activity, service, receiver, provider, activity-alias).
     */
    private static final List<String> COMPONENT_ALLOWED_VARYING_ATTRS = Arrays.asList(
            ATTR_LABEL,
            ATTR_ICON,
            ATTR_ROUND_ICON,
            ATTR_LOGO,
            ATTR_DESCRIPTION,
            ATTR_THEME
    );

    /** Constructs a new {@link ManifestResourceDetector} */
    public ManifestResourceDetector() {
    }

    // ---- Implements XmlScanner ----

    @Override
    public Collection<String> getApplicableElements() {
        return Arrays.asList(
                TAG_APPLICATION,
                TAG_ACTIVITY,
                NODE_ACTIVITY_ALIAS,
                TAG_SERVICE,
                TAG_RECEIVER,
                TAG_PROVIDER,
                NODE_INTENT,
                NODE_ACTION,
                NODE_CATEGORY,
                NODE_DATA,
                NODE_METADATA,
                NODE_GRANT_URI_PERMISSION,
                NODE_PERMISSION,
                NODE_PERMISSION_GROUP,
                NODE_PERMISSION_TREE,
                NODE_USES_PERMISSION,
                NODE_USES_FEATURE,
                NODE_USES_LIBRARY,
                NODE_USES_SDK,
                NODE_INSTRUMENTATION,
                NODE_MANIFEST
        );
    }

    @Override
    public void visitElement(@NonNull XmlContext context, @NonNull Element element) {
        // Only check manifest files
        if (!context.file.getName().equals(ANDROID_MANIFEST_XML)) {
            return;
        }

        String tagName = element.getTagName();

        // Iterate over all attributes of this element
        org.w3c.dom.NamedNodeMap attributes = element.getAttributes();
        for (int i = 0; i < attributes.getLength(); i++) {
            org.w3c.dom.Node node = attributes.item(i);
            if (!(node instanceof Attr)) {
                continue;
            }
            Attr attr = (Attr) node;
            String value = attr.getValue();

            // Only check resource references
            if (value == null || !value.startsWith("@")) {
                continue;
            }

            // Skip tools namespace attributes
            String namespace = attr.getNamespaceURI();
            if (namespace != null && namespace.equals("http://schemas.android.com/tools")) {
                continue;
            }

            // Parse the resource URL
            ResourceUrl url = ResourceUrl.parse(value);
            if (url == null || url.isFramework()) {
                continue;
            }

            // Check if this attribute is allowed to vary by configuration
            String attrName = attr.getLocalName();
            if (attrName == null) {
                attrName = attr.getName();
                if (attrName != null && attrName.contains(":")) {
                    attrName = attrName.substring(attrName.indexOf(':') + 1);
                }
            }

            boolean allowedToVary = isAllowedToVary(tagName, attrName);
            if (allowedToVary) {
                continue;
            }

            // Check if the resource varies by configuration
            if (resourceVariesByConfiguration(context, url)) {
                Location location = context.getValueLocation(attr);
                String message = String.format(
                        "Resources referenced from the manifest cannot vary by configuration " +
                        "(except for version qualifiers, e.g. `-v21`). Found `%1$s` from " +
                        "element `<%2$s>`, attribute `%3$s`",
                        value, tagName, attr.getName());
                context.report(ISSUE, attr, location, message);
            }
        }
    }

    /**
     * Returns true if the given attribute on the given element is allowed to reference
     * resources that vary by configuration.
     */
    private static boolean isAllowedToVary(@NonNull String tagName, @Nullable String attrName) {
        if (attrName == null) {
            return false;
        }

        switch (tagName) {
            case TAG_APPLICATION:
                return APPLICATION_ALLOWED_VARYING_ATTRS.contains(attrName);
            case TAG_ACTIVITY:
            case NODE_ACTIVITY_ALIAS:
            case TAG_SERVICE:
            case TAG_RECEIVER:
            case TAG_PROVIDER:
                return COMPONENT_ALLOWED_VARYING_ATTRS.contains(attrName);
            default:
                return false;
        }
    }

    /**
     * Returns true if the given resource varies by configuration (other than version qualifiers).
     */
    private static boolean resourceVariesByConfiguration(
            @NonNull XmlContext context,
            @NonNull ResourceUrl url) {

        ResourceRepository repository = context.getClient().getResourceRepository(
                context.getProject(), true, false);
        if (repository == null) {
            return false;
        }

        ResourceType type = url.type;
        String name = url.name;

        List<ResourceItem> items = repository.getResources(ResourceNamespace.TODO(), type, name);
        if (items == null || items.isEmpty()) {
            return false;
        }

        if (items.size() == 1) {
            // Only one configuration, check if it's the default
            ResourceItem item = items.get(0);
            com.android.ide.common.resources.configuration.FolderConfiguration config =
                    item.getConfiguration();
            if (config == null) {
                return false;
            }
            // If it's the default configuration, no variation
            return !isOnlyVersionQualified(config);
        }

        // Multiple items: check if they differ by anything other than version qualifiers
        for (ResourceItem item : items) {
            com.android.ide.common.resources.configuration.FolderConfiguration config =
                    item.getConfiguration();
            if (config == null) {
                continue;
            }
            if (hasNonVersionQualifier(config)) {
                return true;
            }
        }

        return false;
    }

    /**
     * Returns true if the folder configuration has qualifiers other than version qualifiers.
     */
    private static boolean hasNonVersionQualifier(
            @NonNull com.android.ide.common.resources.configuration.FolderConfiguration config) {
        // Get the folder name and check for non-version qualifiers
        String folderName = config.getFolderName(ResourceFolderType.VALUES);
        if (folderName == null || folderName.equals("values")) {
            return false;
        }

        // Strip the "values" prefix
        String qualifiers = folderName.substring("values".length());
        if (qualifiers.isEmpty()) {
            return false;
        }

        // Split by '-' and check each qualifier
        String[] parts = qualifiers.split("-");
        for (String part : parts) {
            if (part.isEmpty()) {
                continue;
            }
            // Version qualifiers look like "v21", "v14", etc.
            if (part.matches("v\\d+")) {
                continue;
            }
            // This is a non-version qualifier
            return true;
        }

        return false;
    }

    /**
     * Returns true if the folder configuration only has version qualifiers (or no qualifiers).
     */
    private static boolean isOnlyVersionQualified(
            @NonNull com.android.ide.common.resources.configuration.FolderConfiguration config) {
        return !hasNonVersionQualifier(config);
    }
}
