    private void checkElement(@NonNull XmlContext context, @NonNull Element element) {
        // Check this element's layout params against its parent
        Node parentNode = element.getParentNode();
        if (parentNode != null && parentNode.getNodeType() == Node.ELEMENT_NODE) {
            Element parent = (Element) parentNode;
            String parentTag = parent.getTagName();

            // Handle <merge> tag - we can't determine the actual parent layout
            if (!VIEW_MERGE.equals(parentTag)) {
                checkLayoutParams(context, element, parentTag);
            }
        }

        // Recurse into children
        Node child = element.getFirstChild();
        while (child != null) {
            if (child.getNodeType() == Node.ELEMENT_NODE) {
                checkElement(context, (Element) child);
            }
            child = child.getNextSibling();
        }
    }

    private void checkLayoutParams(@NonNull XmlContext context, @NonNull Element element,
            @NonNull String parentTag) {
        // Get the set of valid layout params for this parent
        Set<String> validSpecificParams = LAYOUT_TO_PARAMS.get(parentTag);
        boolean supportsMargins = MARGIN_SUPPORTING_LAYOUTS.contains(parentTag);
        boolean knownParent = validSpecificParams != null || MARGIN_SUPPORTING_LAYOUTS.contains(parentTag);

        // If we don't know the parent layout at all, skip
        if (!knownParent) {
            return;
        }

        // Check all attributes of the element
        NamedNodeMap attributes = element.getAttributes();
        for (int i = 0; i < attributes.getLength(); i++) {
            Attr attr = (Attr) attributes.item(i);
            String namespaceURI = attr.getNamespaceURI();

            // Only check android namespace attributes
            if (!ANDROID_URI.equals(namespaceURI)) {
                continue;
            }

            String name = attr.getLocalName();

            // Only check layout_ params
            if (!name.startsWith("layout_")) {
                continue;
            }

            // Common params are always valid
            if (COMMON_PARAMS.contains(name)) {
                continue;
            }

            // Check margin params
            if (MARGIN_PARAMS.contains(name)) {
                if (!supportsMargins) {
                    context.report(ISSUE, attr, context.getLocation(attr),
                            String.format(
                                    "Invalid layout param '%1$s' (not defined by parent layout '%2$s')",
                                    name, parentTag));
                }
                continue;
            }

            // If we don't know the specific params for this layout, skip
            if (validSpecificParams == null) {
                continue;
            }

            // Check if the param is valid for this layout
            if (!validSpecificParams.contains(name)) {
                context.report(ISSUE, attr, context.getLocation(attr),
                        String.format(
                                "Invalid layout param '%1$s' (not defined by parent layout '%2$s')",
                                name, parentTag));
            }
        }
    }
