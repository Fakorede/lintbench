package com.android.tools.lint.checks

import com.android.SdkConstants.ANDROID_URI
import com.android.resources.ResourceFolderType
import com.android.tools.lint.detector.api.Category
import com.android.tools.lint.detector.api.Detector
import com.android.tools.lint.detector.api.Implementation
import com.android.tools.lint.detector.api.Issue
import com.android.tools.lint.detector.api.Scope
import com.android.tools.lint.detector.api.Severity
import com.android.tools.lint.detector.api.XmlContext
import com.android.tools.lint.detector.api.XmlScanner
import org.w3c.dom.Element
import org.w3c.dom.Node

class RestrictionsDetector : Detector(), XmlScanner {

    companion object {
        private const val TAG_RESTRICTIONS = "restrictions"
        private const val TAG_RESTRICTION = "restriction"

        private const val ATTR_RESTRICTION_TYPE = "restrictionType"
        private const val ATTR_TITLE = "title"
        private const val ATTR_ENTRIES = "entries"
        private const val ATTR_ENTRY_VALUES = "entryValues"
        private const val ATTR_KEY = "key"
        private const val ATTR_DEFAULT_VALUE = "defaultValue"
        private const val ATTR_DESCRIPTION = "description"

        private const val TYPE_BOOL = "bool"
        private const val TYPE_STRING = "string"
        private const val TYPE_INTEGER = "integer"
        private const val TYPE_CHOICE = "choice"
        private const val TYPE_MULTI_SELECT = "multi-select"
        private const val TYPE_HIDDEN = "hidden"
        private const val TYPE_BUNDLE = "bundle"
        private const val TYPE_BUNDLE_ARRAY = "bundle_array"

        const val MAX_NESTING_DEPTH = 20
        const val MAX_NUMBER_OF_NESTED_RESTRICTIONS = 1000

        private val VALID_TYPES = setOf(
            TYPE_BOOL,
            TYPE_STRING,
            TYPE_INTEGER,
            TYPE_CHOICE,
            TYPE_MULTI_SELECT,
            TYPE_HIDDEN,
            TYPE_BUNDLE,
            TYPE_BUNDLE_ARRAY
        )

        @JvmField
        val ISSUE = Issue.create(
            id = "ValidRestrictions",
            briefDescription = "Invalid Restrictions Descriptor",
            explanation = """
                Ensures that an application's restrictions XML file is properly formed.

                The restrictions XML file must follow the format described in the \
                `RestrictionsManager` documentation. Each `<restriction>` element must have \
                a `android:key` and `android:restrictionType` attribute. The `restrictionType` \
                must be one of the valid types: `bool`, `string`, `integer`, `choice`, \
                `multi-select`, `hidden`, `bundle`, or `bundle_array`. \
                For `choice` and `multi-select` types, `android:entries` and \
                `android:entryValues` are required.
                """,
            category = Category.CORRECTNESS,
            priority = 5,
            severity = Severity.ERROR,
            moreInfo = "https://developer.android.com/reference/android/content/RestrictionsManager.html",
            implementation = Implementation(
                RestrictionsDetector::class.java,
                Scope.RESOURCE_FILE_SCOPE
            )
        )
    }

    override fun appliesTo(folderType: ResourceFolderType): Boolean {
        return folderType == ResourceFolderType.XML
    }

    override fun getApplicableElements(): Collection<String> {
        return listOf(TAG_RESTRICTIONS, TAG_RESTRICTION)
    }

    override fun visitElement(context: XmlContext, element: Element) {
        when (element.tagName) {
            TAG_RESTRICTIONS -> visitRestrictions(context, element)
            TAG_RESTRICTION -> visitRestriction(context, element)
        }
    }

    private fun visitRestrictions(context: XmlContext, element: Element) {
        // Check that children are only <restriction> elements
        val children = element.childNodes
        for (i in 0 until children.length) {
            val child = children.item(i)
            if (child.nodeType == Node.ELEMENT_NODE) {
                val childElement = child as Element
                if (childElement.tagName != TAG_RESTRICTION) {
                    context.report(
                        ISSUE,
                        childElement,
                        context.getElementLocation(childElement),
                        "Unexpected element `<${childElement.tagName}>` in `<restrictions>`: " +
                                "only `<restriction>` elements are allowed"
                    )
                }
            }
        }

        // Check nesting depth and total number of nested restrictions
        checkNestingDepthAndCount(context, element, 0, intArrayOf(0))
    }

    private fun checkNestingDepthAndCount(
        context: XmlContext,
        element: Element,
        depth: Int,
        totalCount: IntArray
    ) {
        val children = element.childNodes
        for (i in 0 until children.length) {
            val child = children.item(i)
            if (child.nodeType == Node.ELEMENT_NODE) {
                val childElement = child as Element
                if (childElement.tagName == TAG_RESTRICTION) {
                    totalCount[0]++
                    if (totalCount[0] > MAX_NUMBER_OF_NESTED_RESTRICTIONS) {
                        context.report(
                            ISSUE,
                            childElement,
                            context.getElementLocation(childElement),
                            "Too many nested restrictions; maximum is $MAX_NUMBER_OF_NESTED_RESTRICTIONS"
                        )
                        return
                    }
                    if (depth + 1 > MAX_NESTING_DEPTH) {
                        context.report(
                            ISSUE,
                            childElement,
                            context.getElementLocation(childElement),
                            "Nesting depth exceeds maximum of $MAX_NESTING_DEPTH"
                        )
                        return
                    }
                    checkNestingDepthAndCount(context, childElement, depth + 1, totalCount)
                }
            }
        }
    }

    private fun getChildElements(element: Element): List<Element> {
        val result = mutableListOf<Element>()
        val children = element.childNodes
        for (i in 0 until children.length) {
            val child = children.item(i)
            if (child.nodeType == Node.ELEMENT_NODE) {
                result.add(child as Element)
            }
        }
        return result
    }

    private fun visitRestriction(context: XmlContext, element: Element) {
        // Must have android:key
        val key = element.getAttributeNS(ANDROID_URI, ATTR_KEY)
        val hasKey = element.hasAttributeNS(ANDROID_URI, ATTR_KEY)
        if (!hasKey) {
            context.report(
                ISSUE,
                element,
                context.getElementLocation(element),
                "Missing required attribute `android:key`"
            )
        } else if (key.isEmpty()) {
            val keyAttr = element.getAttributeNodeNS(ANDROID_URI, ATTR_KEY)
            context.report(
                ISSUE,
                element,
                if (keyAttr != null) context.getLocation(keyAttr) else context.getElementLocation(element),
                "`android:key` must not be empty"
            )
        }

        // Must have android:restrictionType
        val hasRestrictionType = element.hasAttributeNS(ANDROID_URI, ATTR_RESTRICTION_TYPE)
        val restrictionType = element.getAttributeNS(ANDROID_URI, ATTR_RESTRICTION_TYPE)
        if (!hasRestrictionType) {
            context.report(
                ISSUE,
                element,
                context.getElementLocation(element),
                "Missing required attribute `android:restrictionType`"
            )
            return
        }

        // Validate restrictionType value
        if (restrictionType.isNotEmpty() && !VALID_TYPES.contains(restrictionType)) {
            val typeAttr = element.getAttributeNodeNS(ANDROID_URI, ATTR_RESTRICTION_TYPE)
            context.report(
                ISSUE,
                element,
                if (typeAttr != null) context.getLocation(typeAttr) else context.getElementLocation(element),
                "Invalid `android:restrictionType` value `$restrictionType`. " +
                        "Must be one of: ${VALID_TYPES.joinToString(", ") { "`$it`" }}"
            )
            return
        }

        // For choice and multi-select, entries and entryValues are required
        if (restrictionType == TYPE_CHOICE || restrictionType == TYPE_MULTI_SELECT) {
            if (!element.hasAttributeNS(ANDROID_URI, ATTR_ENTRIES)) {
                context.report(
                    ISSUE,
                    element,
                    context.getElementLocation(element),
                    "Missing required attribute `android:entries` for " +
                            "`android:restrictionType=\"$restrictionType\"`"
                )
            }

            if (!element.hasAttributeNS(ANDROID_URI, ATTR_ENTRY_VALUES)) {
                context.report(
                    ISSUE,
                    element,
                    context.getElementLocation(element),
                    "Missing required attribute `android:entryValues` for " +
                            "`android:restrictionType=\"$restrictionType\"`"
                )
            }
        }

        // For bundle and bundle_array, must have children restriction elements
        if (restrictionType == TYPE_BUNDLE || restrictionType == TYPE_BUNDLE_ARRAY) {
            val childElements = getChildElements(element)
            if (childElements.isEmpty()) {
                context.report(
                    ISSUE,
                    element,
                    context.getElementLocation(element),
                    "`android:restrictionType=\"$restrictionType\"` must have child `<restriction>` elements"
                )
            } else {
                // Check that all children are <restriction> elements
                for (childElement in childElements) {
                    if (childElement.tagName != TAG_RESTRICTION) {
                        context.report(
                            ISSUE,
                            childElement,
                            context.getElementLocation(childElement),
                            "Unexpected element `<${childElement.tagName}>` inside " +
                                    "`<restriction android:restrictionType=\"$restrictionType\">`: " +
                                    "only `<restriction>` elements are allowed"
                        )
                    }
                }
            }

            // bundle and bundle_array must not have defaultValue
            if (element.hasAttributeNS(ANDROID_URI, ATTR_DEFAULT_VALUE)) {
                val defaultValueAttr = element.getAttributeNodeNS(ANDROID_URI, ATTR_DEFAULT_VALUE)
                context.report(
                    ISSUE,
                    element,
                    if (defaultValueAttr != null) context.getLocation(defaultValueAttr) else context.getElementLocation(element),
                    "`android:restrictionType=\"$restrictionType\"` should not have a `android:defaultValue` attribute"
                )
            }
        }

        // title is required for non-hidden, non-bundle, non-bundle_array types
        if (restrictionType != TYPE_HIDDEN &&
            restrictionType != TYPE_BUNDLE &&
            restrictionType != TYPE_BUNDLE_ARRAY
        ) {
            if (!element.hasAttributeNS(ANDROID_URI, ATTR_TITLE)) {
                context.report(
                    ISSUE,
                    element,
                    context.getElementLocation(element),
                    "Missing required attribute `android:title` for " +
                            "`android:restrictionType=\"$restrictionType\"`"
                )
            }
        }

        // hidden type must have a defaultValue
        if (restrictionType == TYPE_HIDDEN) {
            if (!element.hasAttributeNS(ANDROID_URI, ATTR_DEFAULT_VALUE)) {
                context.report(
                    ISSUE,
                    element,
                    context.getElementLocation(element),
                    "Missing required attribute `android:defaultValue` for " +
                            "`android:restrictionType=\"hidden\"`"
                )
            }
        }

        // Validate defaultValue based on type
        if (element.hasAttributeNS(ANDROID_URI, ATTR_DEFAULT_VALUE)) {
            val defaultValue = element.getAttributeNS(ANDROID_URI, ATTR_DEFAULT_VALUE)
            val defaultValueAttr = element.getAttributeNodeNS(ANDROID_URI, ATTR_DEFAULT_VALUE)
            val location = if (defaultValueAttr != null) context.getLocation(defaultValueAttr)
            else context.getElementLocation(element)

            when (restrictionType) {
                TYPE_INTEGER -> {
                    // defaultValue must be a valid integer or an @integer/ resource reference
                    if (!defaultValue.startsWith("@") && !isValidInteger(defaultValue)) {
                        context.report(
                            ISSUE,
                            element,
                            location,
                            "Invalid `android:defaultValue` for `android:restrictionType=\"integer\"`: " +
                                    "`$defaultValue` is not a valid integer"
                        )
                    }
                }
                TYPE_BOOL -> {
                    // defaultValue must be true, false, or a @bool/ resource reference
                    if (!defaultValue.startsWith("@") &&
                        defaultValue != "true" && defaultValue != "false"
                    ) {
                        context.report(
                            ISSUE,
                            element,
                            location,
                            "Invalid `android:defaultValue` for `android:restrictionType=\"bool\"`: " +
                                    "must be `true` or `false`"
                        )
                    }
                }
            }
        }

        // Check for duplicate keys among siblings
        checkDuplicateKeys(context, element)
    }

    private fun isValidInteger(value: String): Boolean {
        return try {
            value.toInt()
            true
        } catch (e: NumberFormatException) {
            try {
                value.toLong()
                true
            } catch (e2: NumberFormatException) {
                false
            }
        }
    }

    private fun checkDuplicateKeys(context: XmlContext, element: Element) {
        val parent = element.parentNode as? Element ?: return
        val key = element.getAttributeNS(ANDROID_URI, ATTR_KEY)
        if (key.isEmpty()) return

        // Only report on the duplicate (not the first occurrence)
        val siblings = parent.childNodes
        var firstOccurrence: Element? = null
        for (i in 0 until siblings.length) {
            val sibling = siblings.item(i)
            if (sibling.nodeType == Node.ELEMENT_NODE) {
                val siblingElement = sibling as Element
                if (siblingElement.tagName == TAG_RESTRICTION) {
                    val siblingKey = siblingElement.getAttributeNS(ANDROID_URI, ATTR_KEY)
                    if (siblingKey == key) {
                        if (firstOccurrence == null) {
                            firstOccurrence = siblingElement
                        } else if (siblingElement === element) {
                            // This is a duplicate
                            val keyAttr = element.getAttributeNodeNS(ANDROID_URI, ATTR_KEY)
                            context.report(
                                ISSUE,
                                element,
                                if (keyAttr != null) context.getLocation(keyAttr) else context.getElementLocation(element),
                                "Duplicate key `$key`"
                            )
                            break
                        }
                    }
                }
            }
        }
    }
}
