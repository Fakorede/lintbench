        return folderType == ResourceFolderType.MANIFEST;
    }

    @Override
    public void visitElement(XmlContext context, Element element) {
        // No-op since we are only interested in the document level.
    }

    @Override
    public void visitDocument(XmlContext context, Document document) {
        List<String> registeredComponents = getRegisteredComponentNames(context);
        for (PsiClass psiClass : context.getModuleScope().getClasses()) {
            if (!psiClass.isInterface() && !psiClass.hasModifierProperty("abstract")) {
                String className = psiClass.getQualifiedName();
                UClass uClass = context.getUastUtils().convert(psiClass);
                if (uClass != null) {
                    boolean isActivityOrServiceOrProvider = isActivity(uClass) || isService(uClass) || isContentProvider(uClass);
                    if (isActivityOrServiceOrProvider && !registeredComponents.contains(className)) {
                        context.report(ISSUE, psiClass, context.getLocation(psiClass), "Component class %s is not registered in the manifest", className);
                    }
                }
            }
        }
    }

    private boolean isActivity(UClass uClass) {
        return uClass.getSuperClasses().stream()
                .anyMatch(superClass -> superClass.getQualifiedName().equals("android.app.Activity"));
    }

    private boolean isService(UClass uClass) {
        return uClass.getSuperClasses().stream()
                .anyMatch(superClass -> superClass.getQualifiedName().equals("android.app.Service"));
    }

    private boolean isContentProvider(UClass uClass) {
        return uClass.getSuperClasses().stream()
                .anyMatch(superClass -> superClass.getQualifiedName().equals("android.content.ContentProvider"));
    }

    private List<String> getRegisteredComponentNames(XmlContext context) {
        List<String> registeredComponents = new ArrayList<>();
        for (Element element : context.getManifest().getElementsByTagName("application")) {
            Element applicationNode = (Element) element;
            addRegisteredComponents(registeredComponents, applicationNode.getElementsByTagName("activity"));
            addRegisteredComponents(registeredComponents, applicationNode.getElementsByTagName("service"));
            addRegisteredComponents(registeredComponents, applicationNode.getElementsByTagName("provider"));
        }
        return registeredComponents;
    }

    private void addRegisteredComponents(List<String> registeredComponents, NodeList nodeList) {
        for (int i = 0; i < nodeList.getLength(); i++) {
            Element componentElement = (Element) nodeList.item(i);
            String componentName = componentElement.getAttribute("android:name");
            if (!componentName.isEmpty()) {
                registeredComponents.add(componentName);
            }
        }
    }

    @Override

        return new UElementHandler() {
            @Override
            public void visitClass(UClass node) {
                // No-op since we are only interested in the document level.
            }
        };
    }
