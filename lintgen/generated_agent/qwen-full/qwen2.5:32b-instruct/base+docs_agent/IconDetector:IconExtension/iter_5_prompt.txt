                String header = new String(bytes, 0, expectedHeader.getBytes().length);
                if (!header.equals(expectedHeader)) {
                    report(context, file);
                }
            }
        } catch (IOException e) {
            // Ignore and continue
        }
    }

    private void report(@NonNull Context context, @NonNull File file) {

        context.report(ISSUE,

    }
