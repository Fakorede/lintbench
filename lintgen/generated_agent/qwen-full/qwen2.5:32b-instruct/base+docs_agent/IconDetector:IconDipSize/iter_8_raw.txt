package com.android.tools.lint.checks;

import com.android.annotations.NonNull;
import com.android.resources.Density;
import com.android.resources.ResourceFolderType;
import com.android.utils.Pair;
import com.android.utils.XmlUtils;
import com.google.common.collect.ImmutableList;
import com.google.common.collect.Lists;

import org.w3c.dom.Element;

import java.util.Collection;
import java.util.Collections;
import java.util.List;

public class IconDetector extends ResourceXmlScanner {

    private static final String ISSUE_NAME = "IconDensityIndependentSize";
    private static final String ISSUE_ID = "IconDensityIndependentSize";
    private static final int ISSUE_SEVERITY = 5; // ERROR
    private static final Category CATEGORY = Category.USABILITY;
    private static final Scope SCOPE = Scope.RESOURCE_FILE_SCOPE;

    public static final Issue ISSUE = Issue.create(
            ISSUE_ID,
            "Checks that all icons provided in multiple densities compute to roughly the same density-independent pixel (dip) size",
            "This check ensures that icons are consistent across different screen densities, avoiding issues where images are placed in the wrong folder or resized inconsistently.",
            CATEGORY, 5, SCOPE,
            new Implementation(IconDetector.class, Scope.RESOURCE_FILE_SCOPE));

    @NonNull
    @Override
    public Collection<String> getApplicableAttributes() {
        return Collections.singletonList("drawable");
    }

    private final List<Pair<Density, Integer>> densitySizes = Lists.newArrayList();

    @Override
    public void visitAttribute(@NonNull Context context, @NonNull Element element, @NonNull String attributeName) {
        String drawableName = getAttributeValue(context, element, attributeName);
        if (drawableName != null && !drawableName.isEmpty()) {
            ResourceFolderType folderType = context.getResourceFolderType();
            Density density = folderType.getDensity();

            int size = getSizeFromElement(element);
            densitySizes.add(Pair.of(density, size));
        }
    }

    @Override
    public void afterChildren(@NonNull Context context, @NonNull Element element) {
        if (!densitySizes.isEmpty()) {
            checkConsistency(context, element);
            densitySizes.clear();
        }
    }

    private int getSizeFromElement(Element element) {
        // Placeholder for actual logic to extract size from the XML element
        return 48; // Example default value
    }

    private void checkConsistency(@NonNull Context context, @NonNull Element element) {
        if (densitySizes.size() < 2) {
            return;
        }

        int baseSize = densitySizes.get(0).second();
        for (Pair<Density, Integer> pair : densitySizes) {
            Density density = pair.first();
            int size = pair.second();

            // Calculate the expected size based on the density
            float expectedDpSize = baseSize * density.getValue();
            if (!isCloseEnough(size, expectedDpSize)) {
                context.report(ISSUE, element, context.getLocation(element),
                        "Icon sizes are not consistent across densities");
                return;
            }
        }
    }

    private boolean isCloseEnough(int actualPxSize, float expectedDpSize) {
        // Placeholder for logic to determine if the size is close enough
        final int TOLERANCE = 5; // Example tolerance value in pixels
        return Math.abs(actualPxSize - (int) expectedDpSize) <= TOLERANCE;
    }
}
