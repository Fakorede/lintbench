package com.android.tools.lint.checks

import com.android.SdkConstants
import com.android.annotations.VisibleForTesting
import com.android.resources.ResourceFolderType
import com.android.tools.lint.detector.api.*
import org.jetbrains.uast.UCallExpression
import org.w3c.dom.Element

class BatteryDetector : Detector(), SourceCodeScanner, XmlScanner {
    companion object Issues {
        val BATTERY_LIFE_ISSUE = Issue.create(
            id = "BatteryLifeIssues",
            briefDescription = "Negative impact on battery life or uses deprecated APIs",
            explanation = """
                This issue flags code that either negatively affects battery life, or uses APIs that have recently changed behavior to prevent background tasks from consuming memory and battery excessively. Generally, you should be using `WorkManager` instead.
                
                For more details on how to update your code, please see https://developer.android.com/topic/performance/background-optimization
            """,
            category = Category.PERFORMANCE,
            priority = 6,
            severity = Severity.WARNING,
            implementation = Implementation(
                BatteryDetector::class.java,
                Scope.JAVA_FILE_SCOPE or Scope.RESOURCE_FILE_SCOPE
            )
        )

        private val BANNED_METHODS = listOf(
            "AlarmManager.set",
            "AlarmManager.setRepeating",
            "PowerManager.wakeLock"
        )
    }

    override fun getApplicableMethodNames(): List<String> {
        return BANNED_METHODS
    }

    override fun visitMethodCall(context: JavaContext, node: UCallExpression) {
        val methodName = context.evaluator.getMethodName(node)
        if (BANNED_METHODS.contains(methodName)) {
            val issue = BATTERY_LIFE_ISSUE.withDescription(
                "Using $methodName can negatively affect battery life. Consider using WorkManager instead."
            )
            context.report(issue, node, context.getLocation(node), issue.getExplanation())
        }
    }

    override fun visitElement(context: XmlContext, element: Element) {
        val tagName = element.tagName
        if (tagName == SdkConstants.TAG_SERVICE && isBackgroundService(element)) {
            val issue = BATTERY_LIFE_ISSUE.withDescription(
                "Using background services can negatively affect battery life. Consider using WorkManager instead."
            )
            context.report(issue, context.getLocation(element), issue.getExplanation())
        }
    }

    @VisibleForTesting
    fun isBackgroundService(element: Element): Boolean {
        val intentFilter = element.getElementsByTagName(SdkConstants.TAG_INTENT_FILTER)
        if (intentFilter.length > 0) {
            for (i in 0 until intentFilter.length) {
                val filterElement = intentFilter.item(i) as Element
                val actionElements = filterElement.getElementsByTagName(SdkConstants.TAG_ACTION)
                for (j in 0 until actionElements.length) {
                    val actionElement = actionElements.item(j) as Element
                    if (actionElement.getAttribute("android:name").contains("ACTION_BOOT_COMPLETED")) {
                        return true
                    }
                }
            }
        }
        return false
    }

    override fun appliesTo(folderType: ResourceFolderType): Boolean {
        return folderType == ResourceFolderType.MANIFEST
    }
}
