package com.android.tools.lint.checks

import com.android.tools.lint.detector.api.*
import org.jetbrains.uast.*
import java.util.*

class IndentationDetector : Detector(), SourceCodeScanner {
    companion object Issues {
        val SUSPICIOUS_INDENTATION = Issue.create(
            "SuspiciousIndentation",
            "Suspicious indentation suggests a grouping that isn't actually there in the code.",
            "This check looks for cases where the indentation suggests a grouping that isn't actually there in the code. A common example of this would be something like:\n" +
                    "\n" +
                    "```kotlin\n" +
                    "if (column > width)\n" +
                    "line++\n" +
                    "column = 0\n" +
                    "```\n" +
                    "\n" +
                    "Here, the `column = 0` line will be executed every single time, not just if the condition is true.",
            Category.CORRECTNESS,
            5,
            Severity.WARNING,
            Implementation(
                IndentationDetector::class.java,
                Scope.JAVA_FILE_SCOPE
            )
        )
    }

    override fun getApplicableMethodNames(): List<String>? {
        return null
    }

    override fun visitClass(context: JavaContext, declaration: UClass) {
        val statements = mutableListOf<UExpression>()
        var previousIndentLevel = -1

        for (body in declaration.body?.children.orEmpty()) {
            if (body is UBlockExpression || body is USimpleNameReferenceExpression) continue
            val indentLevel = getIndentLevel(body)

            if (previousIndentLevel != -1 && indentLevel > previousIndentLevel) {
                statements.add(body)
            } else if (statements.isNotEmpty() && indentLevel <= previousIndentLevel) {
                checkSuspiciousIndentation(context, statements)
                statements.clear()
            }

            previousIndentLevel = indentLevel
        }

        if (statements.isNotEmpty()) {
            checkSuspiciousIndentation(context, statements)
        }
    }

    private fun getIndentLevel(element: UElement): Int {
        val textRange = element.sourcePsi?.textRange ?: return 0
        val lineStartOffset = element.uastParent?.sourcePsi?.text.indexOf('\n', textRange.startOffset - 1) + 1 ?: 0
        var indentLevel = 0

        for (i in lineStartOffset until textRange.startOffset) {
            if (element.sourcePsi?.containingFile?.text[i] == ' ') {
                indentLevel++
            } else {
                break
            }
        }

        return indentLevel / 4 // Assuming 4 spaces per tab
    }

    private fun checkSuspiciousIndentation(context: JavaContext, statements: List<UExpression>) {
        if (statements.size <= 1) return

        val firstStatement = statements.first()
        val lastStatement = statements.last()

        context.report(
            SUSPICIOUS_INDENTATION,
            lastStatement,
            context.getLocation(lastStatement),
            "Suspicious indentation suggests a grouping that isn't actually there in the code."
        )
    }
}
